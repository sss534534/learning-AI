### 检索质量评估指标

| 指标 | 定义 | 适用场景 | 目标值 |
|------|------|----------|--------|
| **Recall@K** | 前K个结果中相关文档比例 | 确保不遗漏关键信息 | >85% |
| **Precision@K** | 前K个结果中真正相关的比例 | 减少无关信息干扰 | >70% |
| **MRR** | 第一个相关文档排名的倒数 | 关注首条结果质量 | >0.7 |
| **NDCG** | 考虑位置权重的累积增益 | 评估排序质量 | >0.8 |
| **Latency P99** | 99%请求的响应时间 | 系统性能保障 | <200ms |

```python
import numpy as np
from sklearn.decomposition import PCA

class VectorCompressor:
    """向量压缩器 - 降低存储和计算成本"""
    
    def __init__(self, target_dim: int = 256):
        self.target_dim = target_dim
        self.pca = None
        self.scaler = None
    
    def fit(self, vectors: np.ndarray):
        """训练压缩模型"""
        # 标准化
        from sklearn.preprocessing import StandardScaler
        self.scaler = StandardScaler()
        vectors_scaled = self.scaler.fit_transform(vectors)
        
        # PCA降维
        self.pca = PCA(n_components=self.target_dim)
        self.pca.fit(vectors_scaled)
        
        # 计算信息保留率
        explained_var = sum(self.pca.explained_variance_ratio_)
        print(f"PCA降维后信息保留率: {explained_var:.2%}")
        
        return self
    
    def compress(self, vectors: np.ndarray) -> np.ndarray:
        """压缩向量"""
        vectors_scaled = self.scaler.transform(vectors)
        return self.pca.transform(vectors_scaled).astype(np.float16)
    
    def decompress(self, compressed: np.ndarray) -> np.ndarray:
        """解压缩（近似还原）"""
        reconstructed = self.pca.inverse_transform(compressed)
        return self.scaler.inverse_transform(reconstructed)
    
    def binary_quantize(self, vectors: np.ndarray) -> np.ndarray:
        """二值量化 - 极致压缩"""
        return np.packbits(vectors > 0).reshape(len(vectors), -1)

# 使用示例
compressor = VectorCompressor(target_dim=256)
compressor.fit(training_vectors)
compressed = compressor.compress(vectors)  # 3072维 → 256维，压缩12倍
```

## 关键词检索增强

### 1. 传统BM25优化

```python
from rank_bm25 import BM25Okapi
import jieba
from typing import List, Dict

class OptimizedBM25:
    """优化的BM25检索器"""
    
    def __init__(self, documents: List[str], language="zh"):
        self.language = language
        self.documents = documents
        self.tokenized_docs = self._tokenize_documents(documents)
        self.bm25 = BM25Okapi(self.tokenized_docs)
        
        # 计算IDF统计信息
        self._compute_idf_stats()
    
    def _tokenize_documents(self, docs: List[str]) -> List[List[str]]:
        """文档分词"""
        tokenized = []
        for doc in docs:
            if self.language == "zh":
                # 中文分词 + 去停用词
                tokens = [t for t in jieba.lcut(doc) 
                         if len(t) > 1 and not self._is_stopword(t)]
            else:
                tokens = doc.lower().split()
            tokenized.append(tokens)
        return tokenized
    
    def _is_stopword(self, token: str) -> bool:
        """判断是否为停用词"""
        stopwords = set(['的', '了', '在', '是', '我', '有', '和', '就', '不', '人', 
                        '都', '一', '一个', '上', '也', '很', '到', '说', '要', '去'])
        return token in stopwords
    
    def _compute_idf_stats(self):
        """计算IDF统计用于分析"""
        self.idf_scores = {}
        for word, idf in self.bm25.idf.items():
            self.idf_scores[word] = idf
    
    def search(self, query: str, top_k: int = 10) -> List[Dict]:
        """执行BM25检索"""
        # 查询分词
        if self.language == "zh":
            query_tokens = [t for t in jieba.lcut(query) if len(t) > 1]
        else:
            query_tokens = query.lower().split()
        
        # 获取分数
        scores = self.bm25.get_scores(query_tokens)
        
        # 返回Top-K结果
        top_indices = np.argsort(scores)[::-1][:top_k]
        results = []
        for idx in top_indices:
            if scores[idx] > 0:
                results.append({
                    "document": self.documents[idx],
                    "score": float(scores[idx]),
                    "index": int(idx)
                })
        
        return results
    
    def explain_score(self, query: str, doc_index: int) -> Dict:
        """解释BM25评分构成"""
        if self.language == "zh":
            query_tokens = [t for t in jieba.lcut(query) if len(t) > 1]
        else:
            query_tokens = query.lower().split()
        
        explanations = []
        for token in query_tokens:
            if token in self.tokenized_docs[doc_index]:
                idf = self.idf_scores.get(token, 0)
                tf = self.tokenized_docs[doc_index].count(token)
                doc_len = len(self.tokenized_docs[doc_index])
                explanations.append({
                    "token": token,
                    "tf": tf,
                    "idf": idf,
                    "doc_len": doc_len
                })
        
        return {
            "query": query,
            "document": self.documents[doc_index][:200],
            "term_explanations": explanations
        }
```

### 2. 稀疏向量检索（SPLADE）

```python
# SPLADE: 学习得到的稀疏表示，结合BM25的可解释性和向量的语义能力
from transformers import AutoModelForMaskedLM, AutoTokenizer
import torch

class SPLADERetriever:
    """SPLADE稀疏向量检索器"""
    
    def __init__(self, model_name="naver/splade-cocondenser-ensembledistil"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForMaskedLM.from_pretrained(model_name)
        self.model.eval()
    
    def encode(self, texts: List[str], batch_size: int = 32) -> np.ndarray:
        """编码文本为稀疏向量"""
        all_embeddings = []
        
        for i in range(0, len(texts), batch_size):
            batch = texts[i:i + batch_size]
            inputs = self.tokenizer(
                batch, 
                return_tensors="pt", 
                padding=True, 
                truncation=True, 
                max_length=512
            )
            
            with torch.no_grad():
                output = self.model(**inputs)
                # 使用logits的relu激活和max pooling
                logits = output.logits
                activations = torch.log(1 + torch.relu(logits))
                # Max pooling over sequence
                embeddings = torch.max(activations, dim=1).values
            
            all_embeddings.append(embeddings.cpu().numpy())
        
        return np.vstack(all_embeddings)
    
    def to_sparse_dict(self, dense_vector: np.ndarray, threshold: float = 0.1) -> Dict:
        """将稠密向量转为稀疏字典表示"""
        # 获取非零元素
        non_zero = np.where(np.abs(dense_vector) > threshold)[0]
        return {
            int(idx): float(dense_vector[idx]) 
            for idx in non_zero
        }
```

## 混合检索策略

### 1. 多路召回架构

```python
from dataclasses import dataclass
from typing import List, Callable
import numpy as np

@dataclass
class RetrievalResult:
    """检索结果"""
    doc_id: str
    content: str
    score: float
    source: str  # 来源：vector/bm25/sparse
    metadata: Dict = None

class HybridRetriever:
    """混合检索器 - 多路召回融合"""
    
    def __init__(self):
        self.retrievers = {}
        self.weights = {}
    
    def add_retriever(self, name: str, retriever: Callable, weight: float = 1.0):
        """添加检索器"""
        self.retrievers[name] = retriever
        self.weights[name] = weight
    
    def search(
        self, 
        query: str, 
        top_k: int = 10,
        fusion_method: str = "rrf"  # rrf/distribution/weighted
    ) -> List[RetrievalResult]:
        """
        执行混合检索
        
        Args:
            query: 查询文本
            top_k: 返回结果数
            fusion_method: 融合方法
                - rrf: Reciprocal Rank Fusion
                - distribution: 基于分布的归一化
                - weighted: 加权求和
        """
        # 收集各路召回结果
        all_results = {}
        for name, retriever in self.retrievers.items():
            results = retriever(query, top_k=top_k*2)  # 多召回一些用于融合
            all_results[name] = results
        
        # 融合结果
        if fusion_method == "rrf":
            return self._rrf_fusion(all_results, top_k)
        elif fusion_method == "distribution":
            return self._distribution_fusion(all_results, top_k)
        else:
            return self._weighted_fusion(all_results, top_k)
    
    def _rrf_fusion(self, all_results: Dict, top_k: int, k: int = 60) -> List[RetrievalResult]:
        """
        Reciprocal Rank Fusion (RRF)
        公式: score = Σ(1 / (k + rank))
        优点：无需调参，对排名敏感而非绝对分数
        """
        doc_scores = {}
        
        for source, results in all_results.items():
            for rank, result in enumerate(results, start=1):
                doc_id = result.doc_id if hasattr(result, 'doc_id') else result.get('id')
                if doc_id not in doc_scores:
                    doc_scores[doc_id] = {
                        'score': 0,
                        'sources': [],
                        'result': result
                    }
                # RRF公式
                rrf_score = 1.0 / (k + rank)
                doc_scores[doc_id]['score'] += rrf_score * self.weights.get(source, 1.0)
                doc_scores[doc_id]['sources'].append(source)
        
        # 排序并返回
        sorted_docs = sorted(
            doc_scores.items(), 
            key=lambda x: x[1]['score'], 
            reverse=True
        )[:top_k]
        
        return [
            RetrievalResult(
                doc_id=doc_id,
                content=data['result'].content if hasattr(data['result'], 'content') else data['result'].get('content'),
                score=data['score'],
                source=f"hybrid({','.join(data['sources'])})",
                metadata={'sources': data['sources']}
            )
            for doc_id, data in sorted_docs
        ]
    
    def _distribution_fusion(self, all_results: Dict, top_k: int) -> List[RetrievalResult]:
        """基于分布的归一化融合 - 适用于分数差异大的场景"""
        doc_scores = {}
        
        for source, results in all_results.items():
            if not results:
                continue
            
            # 提取分数并归一化到[0,1]
            scores = [r.score if hasattr(r, 'score') else r.get('score', 0) for r in results]
            min_score, max_score = min(scores), max(scores)
            score_range = max_score - min_score if max_score > min_score else 1
            
            for i, result in enumerate(results):
                doc_id = result.doc_id if hasattr(result, 'doc_id') else result.get('id')
                normalized_score = (scores[i] - min_score) / score_range
                
                if doc_id not in doc_scores:
                    doc_scores[doc_id] = {'score': 0, 'result': result, 'sources': []}
                
                doc_scores[doc_id]['score'] += normalized_score * self.weights.get(source, 1.0)
                doc_scores[doc_id]['sources'].append(source)
        
        sorted_docs = sorted(
            doc_scores.items(), 
            key=lambda x: x[1]['score'], 
            reverse=True
        )[:top_k]
        
        return [
            RetrievalResult(
                doc_id=doc_id,
                content=data['result'].content if hasattr(data['result'], 'content') else data['result'].get('content'),
                score=data['score'],
                source=f"hybrid({','.join(data['sources'])})",
                metadata={'sources': data['sources']}
            )
            for doc_id, data in sorted_docs
        ]
```

### 2. 检索结果重排序

```python
from transformers import AutoTokenizer, AutoModelForSequenceClassification
import torch
import torch.nn.functional as F

class Reranker:
    """检索结果重排序器"""
    
    def __init__(self, model_name="BAAI/bge-reranker-large"):
        self.tokenizer = AutoTokenizer.from_pretrained(model_name)
        self.model = AutoModelForSequenceClassification.from_pretrained(model_name)
        self.model.eval()
        self.max_length = 512
    
    def rerank(
        self, 
        query: str, 
        documents: List[str], 
        batch_size: int = 8
    ) -> List[Dict]:
        """
        对检索结果进行重排序
        
        Returns:
            按相关性排序的文档列表，包含重排序分数
        """
        pairs = [[query, doc] for doc in documents]
        scores = []
        
        for i in range(0, len(pairs), batch_size):
            batch_pairs = pairs[i:i + batch_size]
            
            inputs = self.tokenizer(
                batch_pairs,
                padding=True,
                truncation=True,
                max_length=self.max_length,
                return_tensors="pt"
            )
            
            with torch.no_grad():
                outputs = self.model(**inputs)
                batch_scores = F.softmax(outputs.logits, dim=-1)[:, 1]  # 取正类概率
                scores.extend(batch_scores.cpu().numpy())
        
        # 组合结果并排序
        results = [
            {
                'document': doc,
                'rerank_score': float(score),
                'original_index': i
            }
            for i, (doc, score) in enumerate(zip(documents, scores))
        ]
        
        return sorted(results, key=lambda x: x['rerank_score'], reverse=True)
    
    def rerank_with_threshold(
        self, 
        query: str, 
        documents: List[str], 
        threshold: float = 0.5,
        top_k: int = 5
    ) -> List[Dict]:
        """带阈值过滤的重排序"""
        results = self.rerank(query, documents)
        
        # 过滤低分结果
        filtered = [r for r in results if r['rerank_score'] >= threshold]
        
        # 如果没有结果通过阈值，返回Top1
        if not filtered and results:
            return [results[0]]
        
        return filtered[:top_k]

# 级联检索流程
class CascadeRetriever:
    """级联检索：粗排 → 精排 → 重排序"""
    
    def __init__(
        self, 
        coarse_retriever,  # 快速粗排（如HNSW）
        fine_retriever,    # 精确检索（如暴力搜索）
        reranker: Reranker
    ):
        self.coarse_retriever = coarse_retriever
        self.fine_retriever = fine_retriever
        self.reranker = reranker
    
    def search(self, query: str, final_k: int = 5) -> List[Dict]:
        """级联检索流程"""
        # Stage 1: 粗排 - 快速召回大量候选
        coarse_results = self.coarse_retriever.search(query, top_k=100)
        print(f"粗排召回: {len(coarse_results)} 条")
        
        # Stage 2: 精排 - 在候选集上做精确检索
        candidate_ids = [r.doc_id for r in coarse_results]
        fine_results = self.fine_retriever.search(
            query, 
            candidate_ids=candidate_ids,
            top_k=20
        )
        print(f"精排筛选: {len(fine_results)} 条")
        
        # Stage 3: 重排序 - 使用Cross-Encoder精排
        documents = [r.content for r in fine_results]
        reranked = self.reranker.rerank(query, documents, top_k=final_k)
        print(f"重排序后: {len(reranked)} 条")
        
        return reranked
```

## 查询理解与改写

### 1. 查询扩展

```python
from openai import OpenAI
import json

class QueryExpander:
    """查询扩展器 - 生成同义/相关查询"""
    
    def __init__(self, api_key: str):
        self.client = OpenAI(api_key=api_key)
    
    def expand(
        self, 
        query: str, 
        method: str = "hyde",  # hyde/synonym/multi_query
        num_expansions: int = 3
    ) -> List[str]:
        """扩展查询"""
        if method == "hyde":
            return self._hyde_expansion(query, num_expansions)
        elif method == "synonym":
            return self._synonym_expansion(query, num_expansions)
        elif method == "multi_query":
            return self._multi_query_expansion(query, num_expansions)
        else:
            return [query]
    
    def _hyde_expansion(self, query: str, num: int) -> List[str]:
        """
        HyDE (Hypothetical Document Embedding)
        生成假设的答案文档，然后用答案做检索
        """
        prompt = f"""基于以下问题，生成{num}个可能的答案段落。
        每个段落应该详细且包含关键信息。
        
        问题: {query}
        
        以JSON格式返回: {{"answers": ["答案1", "答案2", ...]}}"""
        
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"}
        )
        
        result = json.loads(response.choices[0].message.content)
        return [query] + result.get("answers", [])
    
    def _synonym_expansion(self, query: str, num: int) -> List[str]:
        """同义词扩展"""
        prompt = f"""为以下查询生成{num}个语义等价的改写版本。
        保持原意不变，使用不同的表达方式。
        
        原查询: {query}
        
        以JSON格式返回: {{"variations": ["改写1", "改写2", ...]}}"""
        
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"}
        )
        
        result = json.loads(response.choices[0].message.content)
        return [query] + result.get("variations", [])
    
    def _multi_query_expansion(self, query: str, num: int) -> List[str]:
        """多角度查询扩展"""
        prompt = f"""将以下问题分解为{num}个不同的子问题，
        从不同角度探索原问题。
        
        原问题: {query}
        
        以JSON格式返回: {{"sub_queries": ["子问题1", "子问题2", ...]}}"""
        
        response = self.client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": prompt}],
            response_format={"type": "json_object"}
        )
        
        result = json.loads(response.choices[0].message.content)
        return [query] + result.get("sub_queries", [])

# 查询扩展检索
class ExpandedQueryRetriever:
    """基于查询扩展的检索器"""
    
    def __init__(self, base_retriever, expander: QueryExpander):
        self.base_retriever = base_retriever
        self.expander = expander
    
    def search(self, query: str, top_k: int = 10) -> List[RetrievalResult]:
        """使用扩展查询进行检索"""
        # 扩展查询
        expanded_queries = self.expander.expand(query, method="multi_query", num_expansions=3)
        
        # 对每个扩展查询进行检索
        all_results = []
        for eq in expanded_queries:
            results = self.base_retriever.search(eq, top_k=top_k//2)
            for r in results:
                r.metadata = r.metadata or {}
                r.metadata['original_query'] = query
                r.metadata['expanded_query'] = eq
            all_results.extend(results)
        
        # 去重并重新排序
        seen = set()
        unique_results = []
        for r in all_results:
            if r.doc_id not in seen:
                seen.add(r.doc_id)
                unique_results.append(r)
        
        return unique_results[:top_k]
```

```python
import hashlib
import time
from functools import lru_cache
import redis

class RetrievalCache:
    """检索结果缓存"""
    
    def __init__(self, redis_client=None, ttl: int = 3600):
        self.redis = redis_client
        self.local_cache = {}
        self.ttl = ttl
        self.hit_count = 0
        self.miss_count = 0
    
    def _get_cache_key(self, query: str, **kwargs) -> str:
        """生成缓存键"""
        key_data = f"{query}:{sorted(kwargs.items())}"
        return hashlib.md5(key_data.encode()).hexdigest()
    
    def get(self, query: str, **kwargs) -> Optional[List]:
        """获取缓存结果"""
        key = self._get_cache_key(query, **kwargs)
        
        # 先查本地缓存
        if key in self.local_cache:
            data, expiry = self.local_cache[key]
            if time.time() < expiry:
                self.hit_count += 1
                return data
            else:
                del self.local_cache[key]
        
        # 再查Redis
        if self.redis:
            cached = self.redis.get(f"retrieval:{key}")
            if cached:
                self.hit_count += 1
                import json
                return json.loads(cached)
        
        self.miss_count += 1
        return None
    
    def set(self, query: str, results: List, **kwargs):
        """设置缓存"""
        key = self._get_cache_key(query, **kwargs)
        
        # 本地缓存
        self.local_cache[key] = (results, time.time() + self.ttl)
        
        # Redis缓存
        if self.redis:
            import json
            self.redis.setex(
                f"retrieval:{key}",
                self.ttl,
                json.dumps(results, default=str)
            )
    
    def get_stats(self) -> Dict:
        """获取缓存统计"""
        total = self.hit_count + self.miss_count
        hit_rate = self.hit_count / total if total > 0 else 0
        return {
            "hit_count": self.hit_count,
            "miss_count": self.miss_count,
            "hit_rate": f"{hit_rate:.2%}",
            "local_cache_size": len(self.local_cache)
        }

class SemanticCache:
    """语义缓存 - 相似查询复用结果"""
    
    def __init__(self, embedding_model, similarity_threshold: float = 0.95):
        self.embedding_model = embedding_model
        self.threshold = similarity_threshold
        self.cache_entries = []  # [(embedding, results, timestamp)]
```

```python
import asyncio
import aiohttp
from concurrent.futures import ThreadPoolExecutor

class AsyncRetriever:
    """异步检索器"""
    
    def __init__(self, max_workers: int = 10):
        self.max_workers = max_workers
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    async def search_parallel(
        self, 
        queries: List[str], 
        retriever_func,
        **kwargs
    ) -> List[List]:
        """并行执行多个查询"""
        loop = asyncio.get_event_loop()
        
        # 创建任务
        tasks = [
            loop.run_in_executor(
                self.executor, 
                retriever_func, 
                query, 
                **kwargs
            )
            for query in queries
        ]
        
        # 等待所有完成
        results = await asyncio.gather(*tasks, return_exceptions=True)
        
        # 处理异常
        processed = []
        for r in results:
            if isinstance(r, Exception):
                processed.append([])
            else:
                processed.append(r)
        
        return processed
    
    async def hybrid_search_async(
        self,
        query: str,
        retrievers: Dict[str, Callable],
        top_k: int = 10
    ) -> List[RetrievalResult]:
        """异步执行多路召回"""
        loop = asyncio.get_event_loop()
        
        # 并行执行所有检索器
        tasks = {
            name: loop.run_in_executor(self.executor, retriever, query, top_k*2)
            for name, retriever in retrievers.items()
        }
        
        # 等待结果
        results = {}
        for name, task in tasks.items():
            try:
                results[name] = await task
            except Exception as e:
                print(f"检索器 {name} 失败: {e}")
                results[name] = []
        
        # 融合结果
        return self._fuse_results(results, top_k)
    
    def _fuse_results(self, all_results: Dict, top_k: int) -> List[RetrievalResult]:
        """融合多路结果（简化版RRF）"""
        doc_scores = {}
        k = 60
        
        for source, results in all_results.items():
            for rank, result in enumerate(results, 1):
                doc_id = result.get('id') if isinstance(result, dict) else result.doc_id
                if doc_id not in doc_scores:
                    doc_scores[doc_id] = {'score': 0, 'result': result}
                doc_scores[doc_id]['score'] += 1.0 / (k + rank)
        
        sorted_docs = sorted(
            doc_scores.items(),
            key=lambda x: x[1]['score'],
            reverse=True
        )[:top_k]
        
        return [
            RetrievalResult(
                doc_id=doc_id,
                content=data['result'].get('content') if isinstance(data['result'], dict) else data['result'].content,
                score=data['score'],
                source='hybrid'
            )
            for doc_id, data in sorted_docs
        ]
```

```python
class OptimizedRAGPipeline:
    """优化的RAG检索Pipeline"""
    
    def __init__(
        self,
        vector_store,
        bm25_retriever,
        reranker: Reranker,
        query_analyzer: QueryAnalyzer,
        cache: RetrievalCache,
        config: Dict = None
    ):
        self.vector_store = vector_store
        self.bm25_retriever = bm25_retriever
        self.reranker = reranker
        self.query_analyzer = query_analyzer
        self.cache = cache
        self.config = config or {}
    
    def retrieve(self, query: str, **kwargs) -> Dict:
        """
        执行完整的检索流程
        
        Pipeline:
        1. 查询分析 → 2. 缓存检查 → 3. 多路召回 → 4. 融合 → 5. 重排序
        """
        start_time = time.time()
        
        # 1. 查询分析
        analysis = self.query_analyzer.analyze(query)
        retrieval_params = self.query_analyzer.get_retrieval_params(analysis)
        
        # 2. 检查缓存
        cached = self.cache.get(query, **kwargs)
        if cached:
            return {
                "results": cached,
                "from_cache": True,
                "analysis": analysis,
                "time_ms": (time.time() - start_time) * 1000
            }
        
        # 3. 多路召回
        top_k = kwargs.get('top_k', retrieval_params.get('top_k', 10))
        
        # 向量检索
        vector_results = self.vector_store.search(
            query, 
            top_k=top_k*2,
            filters=kwargs.get('filters')
        )
        
        # BM25检索
        bm25_results = self.bm25_retriever.search(query, top_k=top_k*2)
        
        # 4. 结果融合
        hybrid_retriever = HybridRetriever()
        hybrid_retriever.add_retriever("vector", lambda q, k: vector_results, weight=1.0)
        hybrid_retriever.add_retriever("bm25", lambda q, k: bm25_results, weight=0.8)
        
        fused_results = hybrid_retriever.search(query, top_k=top_k*2)
        
        # 5. 重排序
        if retrieval_params.get('rerank', True):
            documents = [r.content for r in fused_results]
            reranked = self.reranker.rerank(query, documents, top_k=top_k)
            
            # 组装最终结果
            final_results = []
            for r in reranked:
                original = fused_results[r['original_index']]
                final_results.append({
                    "doc_id": original.doc_id,
                    "content": r['document'],
                    "score": r['rerank_score'],
                    "sources": original.metadata.get('sources', []) if original.metadata else []
                })
        else:
            final_results = [
                {
                    "doc_id": r.doc_id,
                    "content": r.content,
                    "score": r.score,
                    "sources": r.metadata.get('sources', []) if r.metadata else []
                }
                for r in fused_results[:top_k]
            ]
        
        # 写入缓存
        self.cache.set(query, final_results, **kwargs)
        
        elapsed_ms = (time.time() - start_time) * 1000
        
        return {
            "results": final_results,
            "from_cache": False,
            "analysis": analysis,
            "retrieval_params": retrieval_params,
            "time_ms": elapsed_ms,
            "stages": {
                "vector_count": len(vector_results),
                "bm25_count": len(bm25_results),
                "fused_count": len(fused_results),
                "final_count": len(final_results)
            }
        }

# 使用示例
```

### 性能监控与调优

```python
class RetrievalMonitor:
    """检索性能监控"""
    
    def __init__(self):
        self.metrics = {
            "query_count": 0,
            "total_latency_ms": 0,
            "cache_hits": 0,
            "stage_latencies": {
                "analysis": [],
                "vector_search": [],
                "bm25_search": [],
                "fusion": [],
                "rerank": []
            }
        }
    
    def record(self, stage: str, latency_ms: float):
        """记录阶段耗时"""
        if stage in self.metrics["stage_latencies"]:
            self.metrics["stage_latencies"][stage].append(latency_ms)
    
    def record_query(self, latency_ms: float, from_cache: bool):
        """记录查询"""
        self.metrics["query_count"] += 1
        self.metrics["total_latency_ms"] += latency_ms
        if from_cache:
            self.metrics["cache_hits"] += 1
    
    def get_report(self) -> Dict:
        """生成性能报告"""
        query_count = self.metrics["query_count"]
        
        report = {
            "summary": {
                "total_queries": query_count,
                "avg_latency_ms": self.metrics["total_latency_ms"] / query_count if query_count > 0 else 0,
                "cache_hit_rate": self.metrics["cache_hits"] / query_count if query_count > 0 else 0
            },
            "stage_analysis": {}
        }
        
        for stage, latencies in self.metrics["stage_latencies"].items():
            if latencies:
                report["stage_analysis"][stage] = {
                    "avg_ms": sum(latencies) / len(latencies),
                    "p50_ms": sorted(latencies)[len(latencies)//2],
                    "p99_ms": sorted(latencies)[int(len(latencies)*0.99)],
                    "total_calls": len(latencies)
                }
        
        return report
    
    def identify_bottleneck(self) -> List[str]:
        """识别性能瓶颈"""
        bottlenecks = []
        
        for stage, data in self.get_report()["stage_analysis"].items():
            if data["avg_ms"] > 100:  # 超过100ms认为有优化空间
                bottlenecks.append(f"{stage}: {data['avg_ms']:.1f}ms avg")
        
        return bottlenecks
```

## 深度分析

检索策略是RAG系统的核心骨架，其质量直接决定了最终生成答案的上限。传统检索往往只依赖单一向量相似度，但生产环境中的查询多样性要求我们必须构建多策略融合的检索体系。从向量检索到BM25关键词匹配，再到SPLADE稀疏向量和知识图谱检索，不同策略在不同查询类型上各有优劣，关键在于如何通过自适应权重将它们有机整合。

查询理解与改写是检索优化中最具杠杆效应的环节。HyDE方法通过生成假设性文档来弥合查询与文档之间的语义鸿沟，查询扩展则从多个角度覆盖用户意图。结合意图识别和查询分析，系统可以动态调整检索参数、选择最合适的检索策略组合，这种智能化的检索前处理往往能带来Recall@K提升10-20%的显著效果。

级联检索架构（粗排→精排→重排序）在生产环境中展现出最佳的性能-精度平衡。粗排阶段使用HNSW等快速索引从海量候选集中筛选出Top-K，精排阶段在缩小后的候选集上做更精确的匹配，最后通过Cross-Encoder重排序器对结果进行精细排序。这种架构通过分层减少计算量，使P99延迟控制在200ms以内的同时保持90%以上的检索精度。
