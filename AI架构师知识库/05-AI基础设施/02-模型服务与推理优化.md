## 推理引擎选型

### 主流推理引擎对比

| 引擎 | 适用模型 | 核心特性 | 性能特点 | 部署复杂度 |
|------|----------|----------|----------|------------|
| **vLLM** | GPT/LLaMA/Qwen等 | PagedAttention、连续批处理 | 高吞吐、低延迟 | 低 |
| **TGI** | HuggingFace模型 | FlashAttention、Safetensors | 稳定、生态好 | 低 |
| **TensorRT-LLM** | NVIDIA GPU优化 | FP8/INT8量化、Kernel融合 | 极致性能 | 高 |
| **llama.cpp** | GGUF格式 | CPU推理、量化支持 | 边缘部署 | 低 |
| **ONNX Runtime** | 通用ONNX | 跨平台、图优化 | 通用性强 | 中 |
| **DeepSpeed** | 大模型训练/推理 | ZeRO、模型并行 | 超大规模 | 高 |
| **SGLang** | 结构化生成 | RadixAttention、DSL | 结构化输出快 | 中 |

### 1. 内存优化

```python
class MemoryOptimizer:
    """内存优化策略"""
    
    @staticmethod
    def calculate_gpu_memory(
        model_params: int,  # 参数量
        batch_size: int,
        seq_length: int,
        dtype_bytes: int = 2,  # FP16=2, FP32=4
        kv_cache_factor: float = 2.0
    ) -> dict:
        """计算GPU内存需求"""
        
        # 模型权重内存
        model_memory = model_params * dtype_bytes / (1024**3)  # GB
        
        # KV Cache内存
        # 每层: 2 (K+V) * num_heads * head_dim * seq_len * batch_size * dtype_bytes
        # 简化估算
        kv_cache_memory = (
            batch_size * seq_length * model_params * 0.001 * kv_cache_factor / (1024**3)
        )
        
        # 激活内存 (粗略估算)
        activation_memory = batch_size * seq_length * 0.001  # GB
        
        # 开销 (CUDA context, fragmentation等)
        overhead = 2.0  # GB
        
        total = model_memory + kv_cache_memory + activation_memory + overhead
        
        return {
            "model_weights_gb": round(model_memory, 2),
            "kv_cache_gb": round(kv_cache_memory, 2),
            "activation_gb": round(activation_memory, 2),
            "overhead_gb": overhead,
            "total_gb": round(total, 2),
            "recommended_gpu_memory_gb": round(total * 1.15, 2)  # 15% buffer
        }
    
    @staticmethod
    def optimize_kv_cache(
        max_batch_size: int = 64,
        max_seq_len: int = 8192,
        block_size: int = 16
    ):
        """优化KV Cache配置"""
        
        # PagedAttention块大小调优
        # 较小的block_size: 内存利用率高，但管理开销大
        # 较大的block_size: 管理简单，但可能有内部碎片
        
        if max_seq_len <= 2048:
            recommended_block_size = 16
        elif max_seq_len <= 8192:
            recommended_block_size = 32
        else:
            recommended_block_size = 64
        
        # 计算最大块数
        max_num_blocks = (max_batch_size * max_seq_len) // recommended_block_size
        
        return {
            "block_size": recommended_block_size,
            "max_num_blocks": max_num_blocks,
            "gpu_memory_utilization": 0.85,
            "swap_space_gb": 4
        }

# 显存优化实践
def apply_memory_optimizations(model_config: dict):
    """应用内存优化"""
    
    optimizations = []
    
    # 1. 梯度检查点 (训练时)

        optimizations.append("gradient_checkpointing")
    
    # 2. 混合精度

    # 3. 序列并行 (长序列)

        optimizations.append("sequence_parallelism")

    if model_config.get("large_model", False):
        optimizations.append("activation_recomputation")
    
    return optimizations
```

## 批处理与动态批处理

### 动态批处理实现

```python
import asyncio
from typing import List, Dict, Callable
from dataclasses import dataclass
import time

@dataclass
class InferenceRequest:
    """推理请求"""
    id: str
    prompt: str
    sampling_params: dict
    future: asyncio.Future
    timestamp: float
    priority: int = 0

class DynamicBatcher:
    """动态批处理器"""
    
    def __init__(
        self,
        inference_func: Callable,
        max_batch_size: int = 32,
        max_wait_ms: float = 10.0,
        max_tokens_per_batch: int = 4096
    ):
        self.inference_func = inference_func
        self.max_batch_size = max_batch_size
        self.max_wait_ms = max_wait_ms
        self.max_tokens_per_batch = max_tokens_per_batch
        
        self.request_queue = asyncio.PriorityQueue()
        self.running = False
        self.batch_stats = {
            "batches_processed": 0,
            "avg_batch_size": 0,
            "avg_latency_ms": 0
        }
    
    async def start(self):
        """启动批处理循环"""
        self.running = True
        while self.running:
            batch = await self._collect_batch()
            if batch:
                asyncio.create_task(self._process_batch(batch))
    
    async def submit(self, request: InferenceRequest) -> asyncio.Future:
        """提交请求"""
        await self.request_queue.put((request.priority, time.time(), request))
        return request.future
    
    async def _collect_batch(self) -> List[InferenceRequest]:
        """收集一批请求"""
        batch = []
        total_tokens = 0
        start_time = time.time()
        
        while len(batch) < self.max_batch_size:
            # 计算剩余等待时间
            elapsed_ms = (time.time() - start_time) * 1000
            remaining_ms = max(0, self.max_wait_ms - elapsed_ms)
            
            if remaining_ms <= 0 and batch:
                break
            
            try:
                _, _, request = await asyncio.wait_for(
                    self.request_queue.get(),
                    timeout=remaining_ms / 1000 if batch else None
                )
                
                # 估算token数 (简化: 字符数/4)
                estimated_tokens = len(request.prompt) // 4
                
                if total_tokens + estimated_tokens > self.max_tokens_per_batch:
                    # 放回队列
                    await self.request_queue.put((request.priority, time.time(), request))
                    break
                
                batch.append(request)
                total_tokens += estimated_tokens
                
            except asyncio.TimeoutError:
                break
        
        return batch
    
    async def _process_batch(self, batch: List[InferenceRequest]):
        """处理批次"""
        if not batch:
            return
        
        start_time = time.time()
        
        try:
            # 提取prompts
            prompts = [req.prompt for req in batch]
            
            # 执行批量推理
            results = await self.inference_func(prompts)
            
            # 分发结果
            for request, result in zip(batch, results):
                if not request.future.done():
                    request.future.set_result(result)
            
            # 更新统计
            latency_ms = (time.time() - start_time) * 1000
            self._update_stats(len(batch), latency_ms)
            
        except Exception as e:
            # 异常处理
            for request in batch:
                if not request.future.done():
                    request.future.set_exception(e)
    
    def _update_stats(self, batch_size: int, latency_ms: float):
        """更新统计信息"""
        n = self.batch_stats["batches_processed"]
        self.batch_stats["batches_processed"] += 1
        self.batch_stats["avg_batch_size"] = (
            (self.batch_stats["avg_batch_size"] * n + batch_size) / (n + 1)
        )
        self.batch_stats["avg_latency_ms"] = (
            (self.batch_stats["avg_latency_ms"] * n + latency_ms) / (n + 1)
        )
    
    def get_stats(self) -> dict:
        """获取统计信息"""
        return dict(self.batch_stats)
    
    def stop(self):
        """停止批处理器"""
        self.running = False
```

### 多级缓存实现

```python
import hashlib
import json
from typing import Optional, List
import redis.asyncio as redis

class PromptCache:
    """Prompt缓存系统"""
    
    def __init__(
        self,
        redis_client: Optional[redis.Redis] = None,
        local_cache_size: int = 1000,
        prefix_cache_enabled: bool = True
    ):
        self.redis = redis_client
        self.local_cache = {}
        self.local_cache_size = local_cache_size
        self.prefix_cache_enabled = prefix_cache_enabled
        self.prefix_tree = {}  # 前缀树
        
        self.stats = {
            "hits": {"local": 0, "redis": 0, "prefix": 0},
            "misses": 0
        }
    
    def _compute_hash(self, prompt: str, params: dict) -> str:
        """计算缓存键"""
        key_data = f"{prompt}:{json.dumps(params, sort_keys=True)}"
        return hashlib.sha256(key_data.encode()).hexdigest()
    
    async def get(self, prompt: str, params: dict) -> Optional[str]:
        """获取缓存结果"""
        cache_key = self._compute_hash(prompt, params)
        
        # 1. 检查本地缓存
        if cache_key in self.local_cache:
            self.stats["hits"]["local"] += 1
            return self.local_cache[cache_key]
        
        # 2. 检查Redis缓存
        if self.redis:
            cached = await self.redis.get(f"llm:cache:{cache_key}")
            if cached:
                self.stats["hits"]["redis"] += 1
                # 回填本地缓存
                self._update_local_cache(cache_key, cached)
                return cached
        
        # 3. 检查Prefix缓存
        if self.prefix_cache_enabled:
            prefix_match = self._find_prefix_match(prompt)
            if prefix_match:
                self.stats["hits"]["prefix"] += 1
                # 只计算剩余部分的推理
                return {"type": "prefix", "match": prefix_match}
        
        self.stats["misses"] += 1
        return None
    
    async def set(
        self, 
        prompt: str, 
        params: dict, 
        result: str,
        ttl_seconds: int = 3600
    ):
        """设置缓存"""
        cache_key = self._compute_hash(prompt, params)
        
        # 更新本地缓存
        self._update_local_cache(cache_key, result)
        
        # 更新Redis缓存
        if self.redis:
            await self.redis.setex(
                f"llm:cache:{cache_key}",
                ttl_seconds,
                result
            )
        
        # 更新Prefix树
        if self.prefix_cache_enabled:
            self._update_prefix_tree(prompt, result)
    
    def _update_local_cache(self, key: str, value: str):
        """更新本地缓存 (LRU策略)"""
        if len(self.local_cache) >= self.local_cache_size:
            # 移除最旧的条目
            oldest = next(iter(self.local_cache))
            del self.local_cache[oldest]
        
        self.local_cache[key] = value
    
    def _find_prefix_match(self, prompt: str) -> Optional[dict]:
        """查找前缀匹配"""
        # 简化实现：查找最长公共前缀
        tokens = prompt.split()
        
        for length in range(len(tokens), 0, -1):
            prefix = " ".join(tokens[:length])
            if prefix in self.prefix_tree:
                return {
                    "prefix": prefix,
                    "remaining": " ".join(tokens[length:]),
                    "cached_result": self.prefix_tree[prefix]
                }
        
        return None
    
    def _update_prefix_tree(self, prompt: str, result: str):
        """更新前缀树"""
        # 存储不同长度的前缀
        tokens = prompt.split()
        for length in [len(tokens), len(tokens)//2, len(tokens)//4]:
            if length > 0:
                prefix = " ".join(tokens[:length])
                self.prefix_tree[prefix] = result
    
    def get_stats(self) -> dict:
        """获取缓存统计"""
        total_hits = sum(self.stats["hits"].values())
        total = total_hits + self.stats["misses"]
        
        return {
            **self.stats,
            "total_requests": total,
            "hit_rate": total_hits / total if total > 0 else 0,
            "local_cache_size": len(self.local_cache),
            "prefix_cache_size": len(self.prefix_tree)
        }

# RadixAttention (vLLM Prefix Caching)
class RadixAttentionCache:
    """RadixAttention前缀缓存实现"""
    
    def __init__(self):
        self.radix_tree = {}
        self.block_size = 16
    
    def compute_prefix_hash(self, tokens: List[int]) -> str:
        """计算前缀哈希"""
        # 按block_size分块哈希
        blocks = [
            tuple(tokens[i:i+self.block_size])
            for i in range(0, len(tokens), self.block_size)
        ]
        
        # 构建Radix树路径
        current = self.radix_tree
        matched_blocks = 0
        
        for block in blocks:
            if block in current:
                current = current[block]
                matched_blocks += 1
            else:
                break
        
        return {
            "matched_blocks": matched_blocks,
            "matched_tokens": matched_blocks * self.block_size,
            "remaining_tokens": tokens[matched_blocks * self.block_size:]
        }
    
    def insert(self, tokens: List[int], kv_cache_blocks: List[int]):
        """插入新的前缀缓存"""
        blocks = [
            tuple(tokens[i:i+self.block_size])
            for i in range(0, len(tokens), self.block_size)
        ]
        
        current = self.radix_tree
        for block in blocks:
            if block not in current:
                current[block] = {}
            current = current[block]
        
        # 存储KV cache块引用
        current["__kv_blocks__"] = kv_cache_blocks
```

```python
class DistributedInference:
    """分布式推理管理"""
    
    def __init__(self, world_size: int, rank: int):

        if not dist.is_initialized():

    @staticmethod
    def setup_tensor_parallel(model: torch.nn.Module, tp_size: int) -> torch.nn.Module:
        """设置张量并行"""
        
        # 使用Megatron-LM或DeepSpeed的并行策略
        from torch.distributed._tensor import DeviceMesh, DTensor
        from torch.distributed.tensor.parallel import parallelize_module
        
        # 创建设备网格
        mesh = DeviceMesh("cuda", list(range(tp_size)))
        
        # 定义并行计划
        parallel_plan = {
            "attention": "colwise_parallel",
            "mlp": "rowwise_parallel"
        }
        
        # 并行化模块
        model = parallelize_module(
            model,
            mesh,
            parallel_plan
        )
        
        return model
    
    @staticmethod
    def setup_pipeline_parallel(
        model: torch.nn.Module, 
        pp_size: int,
        rank: int
    ) -> torch.nn.Module:
        """设置流水线并行"""
        
        # 将模型分层分配到不同GPU
        num_layers = len(model.model.layers)
        layers_per_gpu = num_layers // pp_size
        
        start_layer = rank * layers_per_gpu
        end_layer = start_layer + layers_per_gpu if rank < pp_size - 1 else num_layers
        
        # 只保留当前rank负责的层
        model.model.layers = model.model.layers[start_layer:end_layer]
        
        return model
    
    async def pipeline_inference(
        self,
        input_ids: torch.Tensor,
        num_microbatches: int = 4
    ) -> torch.Tensor:
        """流水线并行推理"""
        
        batch_size = input_ids.shape[0]
        microbatch_size = batch_size // num_microbatches
        
        # 分割成微批次
        microbatches = [
            input_ids[i*microbatch_size:(i+1)*microbatch_size]
            for i in range(num_microbatches)
        ]
        
        outputs = []

            else:
                outputs.append(output)
        
        return torch.cat(outputs, dim=0) if outputs else None

class ExpertParallel:
    """专家并行 - 用于MoE模型"""
    
    def __init__(self, num_experts: int, num_gpus: int):
        self.num_experts = num_experts
        self.num_gpus = num_gpus
        self.experts_per_gpu = num_experts // num_gpus
    
    def route_tokens(
        self, 
        tokens: torch.Tensor,
        router_logits: torch.Tensor,
        top_k: int = 2
    ) -> dict:
        """路由token到专家"""
        
        # 选择Top-K专家
        expert_weights, expert_indices = torch.topk(
            torch.softmax(router_logits, dim=-1),
            top_k,
            dim=-1
        )
        
        # 按GPU分组
        gpu_assignments = {i: [] for i in range(self.num_gpus)}
        
        for batch_idx in range(tokens.shape[0]):
            for k_idx in range(top_k):
                expert_id = expert_indices[batch_idx, k_idx].item()
                gpu_id = expert_id // self.experts_per_gpu
                
                gpu_assignments[gpu_id].append({
                    "token": tokens[batch_idx],
                    "expert": expert_id,
                    "weight": expert_weights[batch_idx, k_idx],
                    "batch_idx": batch_idx
                })
        
        return gpu_assignments
```

### 负载均衡与路由

```python
import random
from typing import List, Dict
import asyncio

class ModelRouter:
    """模型路由器"""
    
    def __init__(self, backends: List[Dict]):
        """
        backends: [
            {"url": "http://vllm-1:8000", "model": "Qwen-72B", "capacity": 256},
            {"url": "http://vllm-2:8000", "model": "Qwen-72B", "capacity": 256}
        ]
        """
        self.backends = backends
        self.backend_stats = {b["url"]: {"load": 0, "healthy": True} for b in backends}
        self.health_check_interval = 30
        
        # 启动健康检查
        asyncio.create_task(self._health_check_loop())
    
    def route(self, request: Dict, strategy: str = "least_loaded") -> str:
        """
        路由请求到后端
        
        Strategies:
        - least_loaded: 最少负载
        - round_robin: 轮询
        - random: 随机
        - model_affinity: 模型亲和性
        """
        healthy_backends = [
            b for b in self.backends 
            if self.backend_stats[b["url"]]["healthy"]
        ]
        
        if not healthy_backends:
            raise Exception("No healthy backend available")
        
        if strategy == "least_loaded":
            return min(
                healthy_backends,
                key=lambda b: self.backend_stats[b["url"]]["load"]
            )["url"]
        
        elif strategy == "round_robin":
            self._rr_index = getattr(self, '_rr_index', 0) + 1
            return healthy_backends[self._rr_index % len(healthy_backends)]["url"]
        
        elif strategy == "random":
            return random.choice(healthy_backends)["url"]
        
        else:
            return healthy_backends[0]["url"]
    
    def update_load(self, backend_url: str, delta: int):
        """更新后端负载"""
        self.backend_stats[backend_url]["load"] += delta
    
    async def _health_check_loop(self):
        """健康检查循环"""
        import aiohttp
        
        while True:
            async with aiohttp.ClientSession() as session:
                for backend in self.backends:
                    try:
                        async with session.get(
                            f"{backend['url']}/health",
                            timeout=aiohttp.ClientTimeout(total=5)
                        ) as resp:
                            self.backend_stats[backend["url"]]["healthy"] = (
                                resp.status == 200
                            )
                    except:
                        self.backend_stats[backend["url"]]["healthy"] = False
            
            await asyncio.sleep(self.health_check_interval)
```

## 深度分析

模型服务架构成熟度直接决定了LLM应用的生产可用性。从单体部署到独立模型服务再到模型网关+多后端，每种架构模式在资源效率、隔离性和灵活性之间做出了不同的权衡。对于生产环境，独立模型服务（如vLLM/TGI）是目前的最佳实践——它实现了应用与模型的解耦，支持独立扩缩容，并可以通过模型网关实现多模型路由、灰度发布和故障切换。

推理引擎的选型是性能优化的核心决策。vLLM凭借PagedAttention和连续批处理成为高吞吐场景的首选，TensorRT-LLM通过FP8量化和算子融合在NVIDIA GPU上达到极致性能，SGLang的RadixAttention在结构化生成场景表现优异。引擎选型不仅取决于模型类型和硬件配置，还与目标延迟SLO密切相关——TTFT<100ms、TPOT<20ms的高要求场景需要更激进的内存优化和并行策略。

内存管理是推理优化的最大挑战。KV Cache随着序列长度和并发数的增长线性膨胀，对显存构成巨大压力。PagedAttention通过分页管理KV Cache解决了内存碎片问题，而Prefix Caching通过复用System Prompt的计算结果避免了重复Prefill。Prefill-Decode分离架构进一步将计算密集的Prefill阶段和内存密集的Decode阶段解耦到不同的GPU池，实现了资源利用率的质的飞跃。
