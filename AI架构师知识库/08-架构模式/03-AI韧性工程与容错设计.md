## 二、熔断器 (Circuit Breaker)

### 2.1 AI专用熔断器

```python
import asyncio
import time
from enum import Enum
from dataclasses import dataclass, field

class CircuitState(Enum):
    CLOSED = "closed"           # 正常
    OPEN = "open"               # 熔断
    HALF_OPEN = "half_open"     # 试探

@dataclass
class AICircuitBreaker:
    """LLM调用专用熔断器"""
    failure_threshold: int = 5        # 连续失败N次→熔断
    timeout_seconds: int = 30         # 熔断持续时间
    half_open_max_requests: int = 3   # 半开状态允许的试探请求
    success_threshold: int = 2        # 连续成功N次→关闭
    
    # 熔断条件(可组合)
    failure_conditions: list = field(default_factory=lambda: [
        "timeout",           # API超时
        "5xx_error",         # 服务端错误
        "rate_limited",      # 被限流
        "empty_response",    # 空响应
        "malformed_output",  # 格式错误
        "hallucination",     # 幻觉率过高
        "toxicity_detected", # 有毒输出
    ])
    
    _state: CircuitState = CircuitState.CLOSED
    _failure_count: int = 0
    _success_count: int = 0
    _last_failure_time: float = 0
    _half_open_requests: int = 0
    
    async def call(self, fn, *args, **kwargs):
        if self._state == CircuitState.OPEN:
            if time.time() - self._last_failure_time > self.timeout_seconds:
                self._state = CircuitState.HALF_OPEN
                self._half_open_requests = 0
            else:
                raise CircuitBreakerOpenError("Circuit is OPEN")
        
        if self._state == CircuitState.HALF_OPEN:
            if self._half_open_requests >= self.half_open_max_requests:
                raise CircuitBreakerOpenError("Half-open limit reached")
            self._half_open_requests += 1
        
        try:
            result = await fn(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure(e)
            raise
    
    def _on_success(self):
        if self._state == CircuitState.HALF_OPEN:
            self._success_count += 1
            if self._success_count >= self.success_threshold:
                self._state = CircuitState.CLOSED
                self._failure_count = 0
                self._success_count = 0
    
    def _on_failure(self, error):
        self._failure_count += 1
        self._last_failure_time = time.time()
        
        if self._failure_count >= self.failure_threshold:
            self._state = CircuitState.OPEN
```

### 2.2 分层熔断策略


## 三、降级策略 (Graceful Degradation)

### 3.1 降级层级

```python
class DegradationLevel(Enum):
    FULL = 1       # 全功能
    REDUCED = 2    # 降功能 (用更小模型)
    MINIMAL = 3    # 最小功能 (缓存+规则)
    OFFLINE = 4    # 离线 (仅错误提示)

class DegradationManager:
    async def generate(self, prompt: str, context: dict):
        try:
            # Level 1: 正常调用
            return await self.call_model("gpt-4o", prompt)
            
        except ModelUnavailableError:
            # Level 2: 降级到便宜模型
            logger.warning("GPT-4o unavailable, falling back to GPT-4o-mini")
            try:
                return await self.call_model("gpt-4o-mini", prompt)
            except ModelUnavailableError:
                pass
            
        except Exception:
            # Level 3: 尝试缓存
            cached = await self.semantic_cache.get(prompt)
            if cached:
                logger.info("Using cached response")
                return cached
            
        # Level 4: 规则引擎兜底
        if self.rule_engine.can_handle(prompt):
            return self.rule_engine.handle(prompt)
        
        # Level 5: 优雅失败
        return self.fallback_response(prompt)
    
    def fallback_response(self, prompt: str) -> str:
        """最后的兜底"""
        return (
            "抱歉，AI服务暂时不可用。"
            "我们的运维团队正在紧急修复中。"
            "预计恢复时间: 15分钟内。\n\n"
            "您可以:\n"
            "1. 稍后重试\n"
            "2. 联系人工客服\n"
            "3. 查看帮助文档"
        )
```

## 四、多模型故障转移

### 4.1 模型路由 + 故障转移

```python
class ResilientModelRouter:
    """带故障转移的模型路由器"""
    
    def __init__(self):
        self.providers = {
            "openai": GPTCircuitBreaker("openai"),
            "anthropic": GPTCircuitBreaker("anthropic"),
            "deepseek": GPTCircuitBreaker("deepseek"),
            "qwen": GPTCircuitBreaker("qwen"),
            "local": LocalModelBreaker("llama-3b"),
        }
        
        # 主备链路
        self.fallback_chains = {
            "gpt-4o": ["gpt-4o-mini", "claude-sonnet-4", "deepseek-chat"],
            "claude-opus-4": ["claude-sonnet-4", "gpt-4o", "deepseek-chat"],
            "deepseek-reasoner": ["deepseek-chat", "qwen2.5-72b"],
        }
    
    async def resilient_call(
        self, 
        messages: list, 
        preferred_model: str,
        max_retries: int = 3
    ) -> dict:
        """自动故障转移调用"""
        chain = [preferred_model] + self.fallback_chains.get(
            preferred_model, []
        )
        
        last_error = None
        for i, model in enumerate(chain[:max_retries + 1]):
            provider = self.get_provider(model)
            breaker = self.providers[provider]
            
            if breaker.state == CircuitState.OPEN:
                logger.info(f"Skipping {model} (circuit open)")
                continue
            
            try:
                result = await breaker.call(
                    self._actual_call, model, messages
                )
                
                # 质量检查
                if not self.quality_check(result, model):
                    logger.warning(f"{model} quality check failed, retrying")
                    continue
                
                return {
                    "result": result,
                    "model_used": model,
                    "fallback_attempts": i
                }
                
            except Exception as e:
                last_error = e
                logger.error(f"{model} failed: {e}")
                continue
        
        raise AllModelsFailedError(
            f"All models failed. Last error: {last_error}"
        )
    
    def quality_check(self, result: dict, model: str) -> bool:
        """输出质量检查"""
        text = result.get("content", "")
        
        # 基础检查
        if not text or len(text) < 1:
            return False
        
        # 幻觉检查(过于自信的错误)
        if "I don't know" in text and len(text) < 50:
            return False  # 过于简短的"I don't know"
        
        # 重复检查
        words = text.split()
        if len(words) > 10:
            unique_ratio = len(set(words)) / len(words)
            if unique_ratio < 0.3:
                return False  # 高度重复
        
        return True
```

### 4.2 健康检查与自动摘除

```python
class ModelHealthChecker:
    """持续健康检查,自动摘除不健康节点"""
    
    async def health_check_loop(self):
        while True:
            for model, instance in self.instances.items():
                health = await self.check_model_health(model, instance)
                
                if health.status == HealthStatus.UNHEALTHY:
                    if instance.consecutive_failures >= 3:
                        logger.error(f"Removing {model} from pool")
                        self.remove_from_pool(model, instance)
                        
                        # 自动告警
                        await self.alert(
                            f"Model {model} removed from serving pool: "
                            f"{health.reason}"
                        )
                
                elif health.status == HealthStatus.DEGRADED:
                    # 降低权重,减少流量
                    self.adjust_weight(model, 0.5)
            
            await asyncio.sleep(10)  # 10秒检查一次
    
    async def check_model_health(
        self, model: str, instance: Instance
    ) -> HealthResult:
        """多为健康检查"""
        checks = []
        
        # 1. 活跃度检查
        checks.append(await self.ping(instance))
        
        # 2. 推理质量(用固定prompt测试)
        checks.append(await self.probe_inference(instance))
        
        # 3. 延迟检查
        checks.append(await self.measure_latency(instance))
        
        # 4. 错误率检查
        checks.append(await self.get_error_rate(instance))
        
        # 综合判断
        if any(c.failed for c in checks):
            return HealthResult(HealthStatus.UNHEALTHY, 
                               reason=str(checks))
        if any(c.degraded for c in checks):
            return HealthResult(HealthStatus.DEGRADED,
                               reason=str(checks))
        return HealthResult(HealthStatus.HEALTHY)
```

```python
class AIChaosExperiment:
    async def run_experiment(
        self, 
        experiment: str,
        duration_seconds: int = 300,
        blast_radius: float = 0.25  # 影响25%流量
    ):
        """自动混沌实验"""

        # 2. 注入故障
        logger.info(f"Starting chaos: {experiment}")
        fault = await self.inject_fault(experiment, blast_radius)
        
        # 3. 观察期
        await asyncio.sleep(duration_seconds)
        
        # 4. 恢复
        await self.remove_fault(fault)
        await asyncio.sleep(30)  # 等待恢复
        
        # 5. 分析结果
        post_experiment = await self.capture_metrics()
        report = self.analyze(baseline, post_experiment, experiment)
        
        # 6. 判断通过/失败
        if report.has_regressions():
            logger.error(f"Chaos experiment FAILED: {experiment}")
            await self.create_incident(report)
        else:
            logger.info(f"Chaos experiment PASSED: {experiment}")
```

## 八、韧性设计模式

### 8.1 模式总结

| 模式 | 问题 | 解决方案 |
|------|------|---------|
| **熔断器** | 级联故障 | 检测失败→快速失败→自动恢复 |
| **舱壁隔离** | 资源争抢 | 按租户/模型分离线程池 |
| **限流** | 过载 | 令牌桶/滑动窗口控制流量 |
| **重试+退避** | 瞬时故障 | 指数退避+抖动 |
| **超时控制** | 资源泄漏 | 严格超时+取消传播 |
| **降级** | 部分故障 | 减少功能保证核心可用 |
| **健康检查** | 慢节点 | 自动摘除+自动恢复 |
| **幂等设计** | 重复请求 | 去重key保证幂等 |

| 级别 | 能力 | 特征 |
|------|------|------|
| **L0: 脆弱** | 无保护 | 任何故障都导致宕机 |
| **L1: 被动** | 基础重试+超时 | 故障后人工介入 |
| **L2: 响应** | 熔断+降级+告警 | 自动检测,人工恢复 |
| **L3: 主动** | 健康检查+自动切换+混沌 | 自愈,预防性 |
| **L4: 弹性** | 全自动+自适应 | 无需人工,故障透明 |

## 深度分析

AI韧性工程与容错设计的核心前提是"故障不可避免"。万卡训练集群的故障数据（GPU硬件故障每天1-2次、网络抖动每小时数次、NCCL超时每天数次）直观展示了大规模AI系统的故障频率。本文从故障模式全景图出发，系统性地覆盖了基础设施、模型服务、依赖服务、流量异常和数据故障五大类故障，为韧性设计提供了完整的威胁模型。

熔断器、降级策略和故障转移是韧性工程的三大支柱。AI专用熔断器相比传统熔断器的增强在于：可组合的失败条件（超时/5xx/限流/空响应/格式错误/幻觉/有毒输出），以及分层熔断策略（单模型→供应商→区域→全局）。降级管理器的五级降级（全功能→降功能→最小功能→规则引擎兜底→优雅失败）确保在任何故障场景下用户都能获得有意义的反馈。`ResilientModelRouter`的故障转移链设计（主模型→Fallback链→质量检查→重试）为生产环境提供了可靠的容错保障。

混沌工程和SLO燃烧率告警是韧性工程的进阶实践。AI混沌实验清单覆盖基础设施、模型服务、依赖服务和流量四个维度，自动化实验框架支持注入故障→观察→恢复→分析的完整流程。多窗口燃烧率告警（1h/6h/24h/72h四个窗口）基于Google SRE方法论，实现了错误预算的精细化监控。韧性成熟度模型（L0脆弱→L4弹性）为企业评估和改进AI系统的韧性水平提供了清晰的路径。
