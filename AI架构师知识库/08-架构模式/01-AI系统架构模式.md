## 7. AI熔断器模式

### 7.1 为什么需要AI熔断器

LLM服务有独特的失败特征：
- **概率性失败**：token级别的不确定性，内容审查拒绝
- **长尾延迟**：P99可能比P50高10倍
- **速率限制**：API提供商有RPM/TPM硬限制
- **计费风险**：部分失败仍消耗token

传统HTTP熔断器（如Resilience4j）无法直接感知"语义级别的失败"。

### 7.2 AI熔断器设计

```python
import time
from enum import Enum
from dataclasses import dataclass, field
from collections import deque

class CircuitState(Enum):
    CLOSED = "closed"              # 正常通行
    OPEN = "open"                  # 拒绝请求
    HALF_OPEN = "half_open"        # 探测恢复

class FailureType(Enum):
    TIMEOUT = "timeout"
    RATE_LIMIT = "rate_limit"
    CONTENT_FILTER = "content_filter"   # 敏感内容拒绝
    HALLUCINATION = "hallucination"     # 幻觉检测触发
    TOKEN_OVERFLOW = "token_overflow"   # 输出过长
    API_ERROR = "api_error"

@dataclass
class AICircuitBreaker:
    """LLM专用熔断器"""

    # 通用配置
    failure_threshold: int = 5        # 连续失败次数阈值
    timeout_seconds: float = 30       # 半开状态超时
    half_open_max_requests: int = 3   # 半开状态最大探测请求数

    # AI特有配置
    content_filter_ratio: float = 0.1 # 内容审查拒绝率阈值
    hallucination_ratio: float = 0.05 # 幻觉率阈值
    p99_latency_threshold_ms: int = 5000  # P99延迟阈值

    # 运行时状态
    state: CircuitState = CircuitState.CLOSED
    failure_count: int = 0
    _recent_failures: deque = field(default_factory=lambda: deque(maxlen=100))
    _recent_latencies: deque = field(default_factory=lambda: deque(maxlen=1000))
    last_failure_time: float = 0

    def before_request(self) -> bool:
        """请求前检查：是否允许通过"""
        if self.state == CircuitState.CLOSED:
            return True

        if self.state == CircuitState.OPEN:
            if time.time() - self.last_failure_time > self.timeout_seconds:
                self.state = CircuitState.HALF_OPEN
                return True
            return False

        # HALF_OPEN: 限制探测请求数
        return len(self._recent_failures) < self.half_open_max_requests

    def on_success(self, latency_ms: float):
        """请求成功回调"""
        self._recent_latencies.append(latency_ms)
        if self.state == CircuitState.HALF_OPEN:
            self.state = CircuitState.CLOSED
            self.failure_count = 0
            self._recent_failures.clear()

    def on_failure(self, failure_type: FailureType, latency_ms: float = 0):
        """请求失败回调"""
        self.failure_count += 1
        self.last_failure_time = time.time()
        self._recent_failures.append({
            "type": failure_type,
            "time": time.time(),
            "latency_ms": latency_ms,
        })

        if failure_type == FailureType.CONTENT_FILTER:
            # 内容审查拒绝：单独计数，更快熔断
            self._check_content_filter_rate()
        elif failure_type == FailureType.HALLUCINATION:
            self._check_hallucination_rate()

        if self.failure_count >= self.failure_threshold:
            self.state = CircuitState.OPEN

    def _check_content_filter_rate(self):
        """检查内容审查拒绝率"""
        recent = list(self._recent_failures)
        filter_fails = sum(
            1 for f in recent
            if f["type"] == FailureType.CONTENT_FILTER
        )
        if len(recent) > 10 and filter_fails / len(recent) > self.content_filter_ratio:
            self.state = CircuitState.OPEN

    def _check_hallucination_rate(self):
        """检查幻觉率"""
        recent = list(self._recent_failures)
        hallucination_count = sum(
            1 for f in recent
            if f["type"] == FailureType.HALLUCINATION
        )
        if len(recent) > 20 and hallucination_count / len(recent) > self.hallucination_ratio:
            self.state = CircuitState.OPEN

# 使用示例
circuit = AICircuitBreaker()

async def safe_llm_call(prompt: str):
    if not circuit.before_request():
        raise CircuitOpenError("LLM熔断器已打开")

    start = time.time()
    try:
        response = await llm.generate(prompt)
        latency = (time.time() - start) * 1000
        circuit.on_success(latency)
        return response
    except ContentFilterError:
        circuit.on_failure(FailureType.CONTENT_FILTER)
        raise
    except RateLimitError:
        circuit.on_failure(FailureType.RATE_LIMIT)
        raise
    except TimeoutError:
        circuit.on_failure(FailureType.TIMEOUT)
        raise
```

## 8. 隔离舱模式 (Bulkhead)

### 8.1 核心思想

将LLM推理资源按业务线或租户物理隔离，防止一个高负载租户拖垮整个服务。

```
                    ┌─────────────────────┐
                    │     API Gateway      │
                    └──────────┬──────────┘
                               │
        ┌──────────────────────┼──────────────────────┐
        │                      │                      │
  ┌─────▼─────┐          ┌─────▼─────┐          ┌─────▼─────┐
  │ Tenant A  │          │ Tenant B  │          │ Tenant C  │
  │ ┌───────┐ │          │ ┌───────┐ │          │ ┌───────┐ │
  │ │ 4 GPU │ │          │ │ 2 GPU │ │          │ │ 1 GPU │ │
  │ │ 队列50│ │          │ │ 队列20│ │          │ │ 队列10│ │
  │ └───────┘ │          │ └───────┘ │          │ └───────┘ │
  └───────────┘          └───────────┘          └───────────┘
```

### 8.2 隔离舱实现

```python
import asyncio
from typing import Dict

class LLMBulkhead:
    """隔离舱：按租户隔离LLM推理资源"""

    def __init__(self):
        self.tenants: Dict[str, TenantPool] = {}

    def configure_tenant(self, name: str, max_concurrency: int,
                         max_queue_size: int, gpu_ids: List[int]):
        self.tenants[name] = TenantPool(
            name=name,
            semaphore=asyncio.Semaphore(max_concurrency),
            queue=asyncio.Queue(maxsize=max_queue_size),
            gpu_ids=gpu_ids,
        )

    async def generate(self, tenant: str, prompt: str) -> str:
        """以租户身份调用LLM"""
        pool = self.tenants.get(tenant)
        if not pool:
            raise ValueError(f"Unknown tenant: {tenant}")

        # 如果队列已满，立即拒绝
        if pool.queue.full():
            raise BulkheadFullError(
                f"Tenant {tenant} bulkhead queue full"
            )

        async with pool.semaphore:
            # 绑定到租户专属GPU
            response = await llm_inference(
                prompt, gpu_devices=pool.gpu_ids
            )
            return response

@dataclass
class TenantPool:
    name: str
    semaphore: asyncio.Semaphore
    queue: asyncio.Queue
    gpu_ids: List[int]

# 配置不同优先级的租户
bulkhead = LLMBulkhead()
bulkhead.configure_tenant("vip_enterprise", max_concurrency=20,
                          max_queue_size=100, gpu_ids=[0, 1, 2, 3])
bulkhead.configure_tenant("standard", max_concurrency=10,
                          max_queue_size=50, gpu_ids=[2, 3])
bulkhead.configure_tenant("free_trial", max_concurrency=5,
                          max_queue_size=20, gpu_ids=[3])
```

## 9. AI限流模式

### 9.1 多维度限流

```python
from dataclasses import dataclass
import time

@dataclass
class AILimiter:
    """LLM多维度限流器"""

    # Token维度
    tokens_per_minute: int = 1_000_000    # TPM
    tokens_per_day: int = 10_000_000      # TPD

    # 请求维度
    requests_per_minute: int = 500        # RPM

    # 成本维度（按美元计）
    cost_per_hour: float = 100.0
    cost_per_day: float = 500.0

    # 并发维度
    max_concurrent: int = 50

    def __post_init__(self):
        self._token_bucket = TokenBucket(
            rate=self.tokens_per_minute, capacity=self.tokens_per_minute
        )
        self._request_bucket = TokenBucket(
            rate=self.requests_per_minute, capacity=self.requests_per_minute
        )
        self._concurrent_sem = asyncio.Semaphore(self.max_concurrent)
        self._cost_window = SlidingWindow(window_seconds=3600)

    async def acquire(self, estimated_tokens: int,
                      estimated_cost: float) -> bool:
        """多维检查，任一不满足则拒绝"""
        checks = [
            self._token_bucket.consume(estimated_tokens),
            self._request_bucket.consume(1),
            self._cost_window.try_add(estimated_cost),
        ]
        return all(checks)


class TokenBucket:
    """令牌桶算法"""
    def __init__(self, rate: int, capacity: int):
        self.rate = rate
        self.capacity = capacity
        self.tokens = capacity
        self.last_refill = time.monotonic()

    def consume(self, n: int) -> bool:
        self._refill()
        if n <= self.tokens:
            self.tokens -= n
            return True
        return False

    def _refill(self):
        now = time.monotonic()
        elapsed = now - self.last_refill
        self.tokens = min(
            self.capacity,
            self.tokens + elapsed * self.rate / 60
        )
        self.last_refill = now
```

### 9.2 自适应限流

```python
class AdaptiveAILimiter(AILimiter):
    """根据系统负载和错误率自适应调整限流阈值"""

    def __init__(self, *args, **kwargs):
        super().__init__(*args, **kwargs)
        self._error_rate = deque(maxlen=100)
        self._p99_latency = deque(maxlen=1000)

    def adaptive_throttle(self) -> float:
        """动态计算限流比例"""
        error_rate = self._current_error_rate()
        p99_latency = self._current_p99()

        # 渐进式限流
        throttle = 0.0
        if error_rate > 0.05:
            throttle = min(0.5, error_rate * 10)   # 5%错误率→50%限流
        if p99_latency > 5000:
            throttle = max(throttle, 0.3)           # P99>5s→30%限流
        if p99_latency > 10000:
            throttle = max(throttle, 0.7)           # P99>10s→70%限流

        return throttle
```

## 10. AI特征开关模式

### 10.1 核心思想

对AI功能的每一次变更（新模型、新Prompt、新策略），使用Feature Flag控制灰度范围，而非全量上线。

```python
from enum import Enum

class AIFeatureFlag:
    """AI特征开关：按用户/比例/属性灰度"""

    def __init__(self, name: str):
        self.name = name
        self.strategies = []

    def add_percentage_strategy(self, percentage: float):
        """按百分比灰度"""
        self.strategies.append(PercentageStrategy(percentage))

    def add_user_list_strategy(self, user_ids: List[str]):
        """按用户白名单灰度"""
        self.strategies.append(UserListStrategy(user_ids))

    def add_attribute_strategy(self, attr: str, values: List[str]):
        """按属性灰度（如企业版用户、付费用户）"""
        self.strategies.append(AttributeStrategy(attr, values))

    def is_enabled(self, context: dict) -> bool:
        """判断该功能是否对该请求生效"""
        return any(s.evaluate(context) for s in self.strategies)

class PercentageStrategy:
    def __init__(self, percentage: float):
        self.percentage = percentage

    def evaluate(self, context):
        user_id = context.get("user_id", "")
        hash_val = hash(user_id) % 100
        return hash_val < self.percentage

# 使用示例：灰度发布新的Prompt模板
feature_flags = {
    "new_prompt_template": AIFeatureFlag("new_prompt_template"),
    "gpt4o_upgrade": AIFeatureFlag("gpt4o_upgrade"),
    "hybrid_search": AIFeatureFlag("hybrid_search"),
}

feature_flags["new_prompt_template"].add_percentage_strategy(10)  # 10%用户
feature_flags["gpt4o_upgrade"].add_attribute_strategy(
    "tier", ["enterprise"]
)  # 仅企业版

# 在请求处理中使用
async def handle_query(user_id: str, tier: str, query: str):
    context = {"user_id": user_id, "tier": tier}

    if feature_flags["new_prompt_template"].is_enabled(context):
        prompt = new_prompt_template.format(query)
    else:
        prompt = old_prompt_template.format(query)

    if feature_flags["gpt4o_upgrade"].is_enabled(context):
        response = await gpt4o.generate(prompt)
    else:
        response = await gpt35.generate(prompt)

    return response
```

## 11. 模型暗启动模式

### 11.1 核心思想

新模型上线时，同时调用新旧模型，但只返回旧模型的结果。新模型的结果用于离线对比分析，无用户影响。

```
用户请求
    │
    ├──► 生产模型 (v1) ──► 返回用户 ✓
    │
    └──► 暗启动模型 (v2) ──► 记录结果 ──► 离线对比
         (不影响用户)
```

### 11.2 暗启动实现

```python
import asyncio

class DarkLaunchManager:
    """模型暗启动管理器"""

    def __init__(self):
        self.dark_models = {}  # {model_version: DarkModelConfig}

    def register_dark_model(self, version: str, model,
                            traffic_pct: float = 1.0,
                            compare_fn=None):
        """注册暗启动模型"""
        self.dark_models[version] = DarkModelConfig(
            model=model,
            traffic_pct=traffic_pct,
            compare_fn=compare_fn or self._default_compare,
        )

    async def process_request(self, user_id: str, prompt: str,
                             production_model) -> str:
        """处理请求：生产响应 + 暗启动对比"""
        # 1. 生产模型：正常响应（阻塞等待）
        production_response = await production_model.generate(prompt)

        # 2. 暗启动模型：异步调用（不阻塞）
        for version, config in self.dark_models.items():
            if self._should_dark_launch(user_id, config.traffic_pct):
                asyncio.create_task(
                    self._dark_compare(version, config, prompt,
                                      production_response)
                )

        return production_response

    async def _dark_compare(self, version: str, config: DarkModelConfig,
                           prompt: str, production_response: str):
        """异步执行暗启动对比"""
        try:
            dark_response = await config.model.generate(prompt)
            comparison = config.compare_fn(
                prompt, production_response, dark_response
            )
            # 记录对比结果用于后续分析
            self._log_comparison(version, comparison)
        except Exception as e:
            self._log_dark_error(version, e)
            # 暗启动异常绝不能影响生产

    def _should_dark_launch(self, user_id: str, pct: float) -> bool:
        return hash(user_id) % 100 < pct * 100

    def _default_compare(self, prompt, prod_resp, dark_resp):
        """默认对比：长度、语义相似度、安全性"""
        return {
            "len_diff": len(dark_resp) - len(prod_resp),
            "semantic_sim": self._semantic_similarity(prod_resp, dark_resp),
            "timestamp": time.time(),
        }
```

class BlueGreenDeployer:
    """AI模型蓝绿部署管理器"""

    def __init__(self, service_registry):
        self.registry = service_registry
        self.active = "blue"
        self.standby = "green"

    async def deploy_new_model(self, model_path: str, config: dict):
        """部署新模型到standby环境"""
        # 1. 部署到standby
        await self.registry.deploy(
            environment=self.standby,
            model_path=model_path,
            config=config,
        )

        # 2. 预热：发送代表性请求，触发模型加载+KV Cache初始化
        await self._warmup(self.standby, config["warmup_queries"])

        # 3. 健康检查
        if not await self._health_check(self.standby):
            raise DeployFailedError("Standby health check failed")

        # 4. 流量切换
        await self.registry.switch_traffic(
            from_env=self.active,
            to_env=self.standby,
            strategy="instant",  # instant | gradual | canary
        )

        # 5. 交换角色
        self.active, self.standby = self.standby, self.active

    async def rollback(self):
        """回滚到上一版本"""
        # 切换回之前的active（现在是standby）
        await self.registry.switch_traffic(
            from_env=self.active,
            to_env=self.standby,
            strategy="instant",
        )
        self.active, self.standby = self.standby, self.active

    async def _warmup(self, env: str, queries: List[str]):
        """预热模型"""
        for query in queries:
            await self.registry.call(env, query)

from typing import Optional, List
from dataclasses import dataclass

@dataclass
class ModelTier:
    name: str
    model: Any
    max_tokens: int
    cost_per_1k: float
    timeout_ms: int
    conditions: List[str] = None  # 触发条件

class CascadeRouter:
    """模型级联路由器"""

    def __init__(self):
        self.tiers: List[ModelTier] = []

    def add_tier(self, tier: ModelTier):
        self.tiers.append(tier)
        # 按优先级排序
        self.tiers.sort(key=lambda t: t.cost_per_1k, reverse=True)

    async def generate(self, prompt: str,
                       max_cost: Optional[float] = None) -> str:
        """级联调用，直到成功或耗尽"""
        errors = []

        for tier in self.tiers:
            # 检查条件
            if tier.conditions:
                if not all(self._check_condition(c) for c in tier.conditions):
                    continue

            try:
                response = await asyncio.wait_for(
                    tier.model.generate(
                        prompt,
                        max_tokens=min(
                            tier.max_tokens,
                            self._estimate_needed_tokens(prompt)
                        )
                    ),
                    timeout=tier.timeout_ms / 1000,
                )
                return response
            except asyncio.TimeoutError:
                errors.append(f"{tier.name}: timeout")
            except RateLimitError:
                errors.append(f"{tier.name}: rate limited")
            except Exception as e:
                errors.append(f"{tier.name}: {e}")

        # 所有tier都失败
        return self._static_fallback()

    def _static_fallback(self) -> str:
        """静态降级回复"""
        return "抱歉，当前系统繁忙，请稍后再试。如有紧急需求，请联系客服。"

# 配置级联
router = CascadeRouter()
router.add_tier(ModelTier(
    name="gpt4o",
    model=gpt4o_model,
    max_tokens=4096,
    cost_per_1k=0.015,
    timeout_ms=15000,
))
router.add_tier(ModelTier(
    name="gpt4o_mini",
    model=gpt4o_mini_model,
    max_tokens=2048,
    cost_per_1k=0.003,
    timeout_ms=10000,
))
router.add_tier(ModelTier(
    name="qwen_local",
    model=qwen_local_model,
    max_tokens=1024,
    cost_per_1k=0.0001,
    timeout_ms=30000,
    conditions=["local_healthy"],  # 仅本地服务健康时使用

class GuardrailPipeline:
    """多层安全护栏管道"""

    def __init__(self):
        self.input_guards = []
        self.output_guards = []

    def add_input_guard(self, guard):
        self.input_guards.append(guard)

    def add_output_guard(self, guard):
        self.output_guards.append(guard)

    async def process(self, user_input: str, user_context: dict) -> dict:
        """完整的安全护栏流程"""
        # 1. 输入护栏
        for guard in self.input_guards:
            result = await guard.check(user_input, user_context)
            if not result.passed:
                return {
                    "blocked": True,
                    "reason": result.reason,
                    "action": result.action,  # "block" | "sanitize" | "warn"
                }

        # 2. LLM推理
        llm_output = await self.llm.generate(user_input)

        # 3. 输出护栏
        for guard in self.output_guards:
            result = await guard.check(llm_output, user_context)
            if not result.passed:
                if result.action == "block":
                    return {
                        "blocked": True,
                        "reason": result.reason,
                        "fallback": "抱歉，我无法回答这个问题。",
                    }
                elif result.action == "sanitize":
                    llm_output = await guard.sanitize(llm_output)

        return {"blocked": False, "output": llm_output}


class PIIOutputGuard:
    """PII泄露检测护栏"""

    async def check(self, output: str, context: dict) -> GuardResult:
        pii_matches = self.pii_detector.detect(output)
        if pii_matches:
            return GuardResult(
                passed=False,
                reason=f"检测到{len(pii_matches)}处PII泄露",
                action="sanitize",
                matches=pii_matches,
            )
        return GuardResult(passed=True)

    async def sanitize(self, output: str) -> str:
        return self.pii_detector.anonymize(output)

class PromptRegistry:
    """Prompt模板注册中心"""

    def __init__(self):
        self.templates = {}      # name → version → template
        self.active_versions = {}  # name → active_version
        self.ab_tests = {}       # name → ABTestConfig

    def register(self, name: str, version: str, template: str):
        if name not in self.templates:
            self.templates[name] = {}
        self.templates[name][version] = template

    def get_prompt(self, name: str, context: dict = None) -> str:
        # A/B测试
        if name in self.ab_tests:
            ab = self.ab_tests[name]
            if ab.should_use_variant(context):
                template = self.templates[name][ab.variant_version]
                return template.format(**context) if context else template

        # 默认Active版本
        version = self.active_versions.get(name, "v1")
        template = self.templates[name][version]
        return template.format(**context) if context else template
