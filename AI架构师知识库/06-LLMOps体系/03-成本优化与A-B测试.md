```python
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import Dict, List, Optional
import json

@dataclass
class CostRecord:
    """成本记录"""
    timestamp: datetime
    model: str
    provider: str
    input_tokens: int
    output_tokens: int
    cost_usd: float
    request_id: str
    user_id: Optional[str] = None
    project_id: Optional[str] = None
    feature: Optional[str] = None

class CostTracker:
    """成本追踪器"""
    
    # 模型定价表 (每1M tokens)
    PRICING = {
        "gpt-4o": {"input": 5.0, "output": 15.0, "cached": 2.5},
        "gpt-4o-mini": {"input": 0.15, "output": 0.60, "cached": 0.075},
        "claude-3-5-sonnet": {"input": 3.0, "output": 15.0},
        "qwen-max": {"input": 2.86, "output": 8.57},  # 转换为USD
        "deepseek-chat": {"input": 0.14, "output": 0.28},
    }
    
    def __init__(self, storage_backend=None):
        self.records: List[CostRecord] = []
        self.storage = storage_backend
        self.daily_budget = 1000.0  # USD
        self.alert_threshold = 0.8
    
    def calculate_cost(
        self,
        model: str,
        input_tokens: int,
        output_tokens: int,
        cached_input: int = 0
    ) -> float:
        """计算单次请求成本"""
        pricing = self.PRICING.get(model, {"input": 5.0, "output": 15.0})
        
        # 正常input
        normal_input = input_tokens - cached_input
        input_cost = (normal_input / 1_000_000) * pricing["input"]
        
        # 缓存input (通常50%折扣)
        if cached_input > 0 and "cached" in pricing:
            input_cost += (cached_input / 1_000_000) * pricing["cached"]
        elif cached_input > 0:
            input_cost += (cached_input / 1_000_000) * pricing["input"] * 0.5
        
        # output
        output_cost = (output_tokens / 1_000_000) * pricing["output"]
        
        return round(input_cost + output_cost, 6)
    
    def record(
        self,
        model: str,
        input_tokens: int,
        output_tokens: int,
        request_id: str,
        user_id: str = None,
        project_id: str = None,
        feature: str = None,
        cached_input: int = 0
    ) -> CostRecord:
        """记录成本"""
        # 推断provider
        provider = self._infer_provider(model)
        
        # 计算成本
        cost = self.calculate_cost(model, input_tokens, output_tokens, cached_input)
        
        record = CostRecord(
            timestamp=datetime.utcnow(),
            model=model,
            provider=provider,
            input_tokens=input_tokens,
            output_tokens=output_tokens,
            cost_usd=cost,
            request_id=request_id,
            user_id=user_id,
            project_id=project_id,
            feature=feature
        )
        
        self.records.append(record)
        
        # 持久化
        if self.storage:
            self.storage.save(record)
        
        # 预算检查
        self._check_budget_alert()
        
        return record
    
    def _infer_provider(self, model: str) -> str:
        """推断模型提供商"""
        if "gpt" in model:
            return "openai"
        elif "claude" in model:
            return "anthropic"
        elif "gemini" in model:
            return "google"
        elif "qwen" in model:
            return "alibaba"
        elif "deepseek" in model:
            return "deepseek"
        return "unknown"
    
    def _check_budget_alert(self):
        """检查预算告警"""
        today_cost = self.get_daily_cost()
        if today_cost > self.daily_budget * self.alert_threshold:
            print(f"⚠️ 预算告警: 今日已使用 ${today_cost:.2f} / ${self.daily_budget}")
    
    def get_daily_cost(self, date: datetime = None) -> float:
        """获取某日成本"""
        date = date or datetime.utcnow()
        start = date.replace(hour=0, minute=0, second=0, microsecond=0)
        end = start + timedelta(days=1)
        
        return sum(
            r.cost_usd for r in self.records
            if start <= r.timestamp < end
        )
    
    def get_cost_breakdown(
        self,
        start: datetime,
        end: datetime,
        group_by: str = "model"
    ) -> Dict:
        """成本分解分析"""
        filtered = [r for r in self.records if start <= r.timestamp < end]
        
        breakdown = {}
        for record in filtered:
            key = getattr(record, group_by, "unknown")
            if key not in breakdown:
                breakdown[key] = {
                    "cost": 0.0,
                    "requests": 0,
                    "input_tokens": 0,
                    "output_tokens": 0
                }
            
            breakdown[key]["cost"] += record.cost_usd
            breakdown[key]["requests"] += 1
            breakdown[key]["input_tokens"] += record.input_tokens
            breakdown[key]["output_tokens"] += record.output_tokens
        
        # 排序
        return dict(sorted(
            breakdown.items(),
            key=lambda x: x[1]["cost"],
            reverse=True
        ))
    
    def get_project_report(self, project_id: str, days: int = 30) -> Dict:
        """项目成本报告"""
        end = datetime.utcnow()
        start = end - timedelta(days=days)
        
        project_records = [
            r for r in self.records
            if r.project_id == project_id and start <= r.timestamp <= end
        ]
        
        total_cost = sum(r.cost_usd for r in project_records)
        total_requests = len(project_records)
        
        # 按模型统计
        by_model = {}
        for r in project_records:
            if r.model not in by_model:
                by_model[r.model] = {"cost": 0, "requests": 0}
            by_model[r.model]["cost"] += r.cost_usd
            by_model[r.model]["requests"] += 1
        
        # 按天统计
        by_day = {}
        for r in project_records:
            day = r.timestamp.strftime("%Y-%m-%d")
            if day not in by_day:
                by_day[day] = 0
            by_day[day] += r.cost_usd
        
        return {
            "project_id": project_id,
            "period_days": days,
            "total_cost_usd": round(total_cost, 2),
            "total_requests": total_requests,
            "avg_cost_per_request": round(total_cost / total_requests, 4) if total_requests > 0 else 0,
            "by_model": by_model,
            "daily_trend": by_day
        }

# 成本分摊计算
class CostAllocation:
    """成本分摊计算器"""
    
    @staticmethod
    def allocate_by_usage(
        total_cost: float,
        usage_by_team: Dict[str, int]  # team -> tokens
    ) -> Dict[str, float]:
        """按使用量分摊"""
        total_usage = sum(usage_by_team.values())
        
        allocation = {}
        for team, usage in usage_by_team.items():
            allocation[team] = round(total_cost * usage / total_usage, 2)
        
        return allocation
    
    @staticmethod
    def allocate_by_project(
        records: List[CostRecord],
        projects: List[str]
    ) -> Dict[str, Dict]:
        """按项目分摊"""
        allocation = {p: {"cost": 0, "tokens": 0, "requests": 0} for p in projects}
        
        for record in records:
            project = record.project_id or "unallocated"
            if project in allocation:
                allocation[project]["cost"] += record.cost_usd
                allocation[project]["tokens"] += record.input_tokens + record.output_tokens
                allocation[project]["requests"] += 1
        
        return allocation
```

```python
from typing import Callable, Dict, List
import random

class SmartRouter:
    """智能模型路由器 - 根据场景选择最优模型"""
    
    def __init__(self, cost_tracker: CostTracker):
        self.cost_tracker = cost_tracker
        self.routing_rules = []
        self.model_performance = {}
    
    def add_rule(
        self,
        condition: Callable[[Dict], bool],
        model: str,
        priority: int = 0
    ):
        """添加路由规则"""
        self.routing_rules.append({
            "condition": condition,
            "model": model,
            "priority": priority
        })
        # 按优先级排序
        self.routing_rules.sort(key=lambda x: -x["priority"])
    
    def route(self, request: Dict) -> str:
        """路由到合适的模型"""
        # 1. 检查规则匹配
        for rule in self.routing_rules:
            if rule["condition"](request):
                return rule["model"]
        
        # 2. 默认路由
        return self._default_route(request)
    
    def _default_route(self, request: Dict) -> str:
        """默认路由策略"""
        prompt = request.get("prompt", "")
        complexity = self._estimate_complexity(prompt)
        
        # 简单任务用轻量级模型
        if complexity == "simple":
            return "gpt-4o-mini"
        elif complexity == "medium":
            return "claude-3-5-sonnet"
        else:
            return "gpt-4o"
    
    def _estimate_complexity(self, prompt: str) -> str:
        """估算任务复杂度"""
        # 基于prompt特征判断
        length = len(prompt)
        
        # 复杂任务关键词
        complex_keywords = [
            "分析", "比较", "评估", "推理", "证明",
            "architecture", "design", "optimize", "analyze"
        ]
        
        has_complex_keywords = any(kw in prompt for kw in complex_keywords)
        
        if length < 100 and not has_complex_keywords:
            return "simple"
        elif length > 1000 or has_complex_keywords:
            return "complex"
        else:
            return "medium"
    
    def setup_cost_aware_routing(self):
        """设置成本感知路由"""
        
        # 规则1: 代码相关任务用Claude
        self.add_rule(
            condition=lambda r: any(kw in r.get("prompt", "") 
                                   for kw in ["代码", "code", "编程", "programming"]),
            model="claude-3-5-sonnet",
            priority=10
        )
        
        # 规则2: 中文任务用Qwen
        self.add_rule(
            condition=lambda r: self._is_chinese(r.get("prompt", "")),
            model="qwen-turbo",
            priority=8
        )
        
        # 规则3: 超长上下文用Gemini
        self.add_rule(
            condition=lambda r: len(r.get("prompt", "")) > 10000,
            model="gemini-1.5-pro",
            priority=9
        )
        
        # 规则4: 简单问答用轻量模型
        self.add_rule(
            condition=lambda r: self._is_simple_qa(r.get("prompt", "")),
            model="gpt-4o-mini",
            priority=5
        )
    
    def _is_chinese(self, text: str) -> bool:
        """检测中文内容"""
        chinese_chars = sum(1 for c in text if '\u4e00' <= c <= '\u9fff')
        return chinese_chars / max(len(text), 1) > 0.3
    
    def _is_simple_qa(self, prompt: str) -> bool:
        """判断是否为简单问答"""
        simple_patterns = [
            "什么是", "什么是", "how to", "what is", "who is",
            "when", "where", "yes/no"
        ]
        return any(p in prompt.lower() for p in simple_patterns)
```

```python
class PromptOptimizer:
    """Prompt优化器 - 减少token消耗"""
    
    def __init__(self, llm_client):
        self.llm_client = llm_client
    
    def compress_prompt(self, prompt: str, target_ratio: float = 0.7) -> str:
        """压缩prompt"""
        original_tokens = self._estimate_tokens(prompt)
        target_tokens = int(original_tokens * target_ratio)
        
        # 移除冗余空白
        compressed = " ".join(prompt.split())
        
        # 如果还太长，使用LLM压缩
        if self._estimate_tokens(compressed) > target_tokens:
            compressed = self._llm_compress(prompt, target_tokens)
        
        return compressed
    
    def _llm_compress(self, prompt: str, target_tokens: int) -> str:
        """使用LLM压缩"""
        compress_prompt = f"""请将以下内容压缩到约{target_tokens}个token，保留所有关键信息：

{prompt}

压缩版本："""
        
        # 调用轻量级模型
        response = self.llm_client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": compress_prompt}],
            max_tokens=target_tokens
        )
        
        return response.choices[0].message.content
    
    def optimize_few_shot(
        self,
        examples: List[Dict],
        max_examples: int = 3
    ) -> List[Dict]:
        """优化few-shot示例"""
        if len(examples) <= max_examples:
            return examples
        
        # 选择最具代表性的示例
        # 策略1: 选择长度适中的
        # 策略2: 覆盖不同场景
        
        # 按长度排序，选择中等的
        sorted_examples = sorted(examples, key=lambda x: len(x.get("input", "")))
        
        # 选择最短、中等、最长的各一个
        selected = [
            sorted_examples[0],
            sorted_examples[len(sorted_examples) // 2],
            sorted_examples[-1]
        ]
        
        return selected[:max_examples]
    
    def summarize_context(
        self,
        context: str,
        max_tokens: int = 500
    ) -> str:
        """总结长上下文"""
        if self._estimate_tokens(context) <= max_tokens:
            return context
        
        summary_prompt = f"""请将以下内容总结到{max_tokens}个token以内：

{context}

总结："""
        
        response = self.llm_client.chat.completions.create(
            model="gpt-4o-mini",
            messages=[{"role": "user", "content": summary_prompt}],
            max_tokens=max_tokens
        )
        
        return response.choices[0].message.content
    
    def _estimate_tokens(self, text: str) -> int:
        """估算token数"""
        # 简化估算: 英文约4字符/token，中文约1.5字符/token
        # 实际应使用tiktoken等库
        return len(text) // 3
    
    @staticmethod
    def remove_redundant_instructions(system_prompt: str) -> str:
        """移除冗余指令"""
        # 常见冗余模式
        redundant_patterns = [
            "请", "请帮我", "能否", "能不能",
            "I want you to", "Please", "Can you"
        ]
        
        cleaned = system_prompt
        for pattern in redundant_patterns:
            cleaned = cleaned.replace(pattern, "")
        
        return cleaned.strip()

class PromptCacheOptimizer:
    """Prompt缓存优化器"""
    
    def __init__(self):
        self.common_prefixes = {}
    
    def extract_common_prefix(self, prompts: List[str]) -> str:
        """提取共同前缀"""
        if not prompts:
            return ""
        
        # 找最长公共前缀
        prefix = prompts[0]
        for prompt in prompts[1:]:
            while not prompt.startswith(prefix):
                prefix = prefix[:-1]
                if not prefix:
```

```python
    def create_experiment(self, config: ExperimentConfig):
        """创建实验"""
        self.experiments[config.experiment_id] = config
        self.results[config.experiment_id] = {"A": [], "B": []}
    
    def assign_variant(self, experiment_id: str, user_id: str) -> str:
        """分配实验组"""
        config = self.experiments[experiment_id]
        
        # 基于user_id哈希确保一致性
        hash_val = int(hashlib.md5(
            f"{experiment_id}:{user_id}".encode()
        ).hexdigest(), 16)
        
        if hash_val % 100 < config.traffic_split * 100:
            return "B"
        else:
            return "A"
    
    def get_model_for_variant(
        self,
        experiment_id: str,
        variant: str
    ) -> str:
        """获取变体对应的模型"""
        config = self.experiments[experiment_id]
        return config.model_b if variant == "B" else config.model_a
    
    def record_result(
        self,
        experiment_id: str,
        variant: str,
        metrics: Dict[str, float],
        cost: float,
        latency_ms: float
    ):
        """记录实验结果"""
        result = {
            "timestamp": datetime.utcnow().isoformat(),
            "metrics": metrics,
            "cost": cost,
            "latency_ms": latency_ms
        }
        
        self.results[experiment_id][variant].append(result)
    
    def analyze_experiment(self, experiment_id: str) -> Dict:
        """分析实验结果"""
        config = self.experiments[experiment_id]
        results_a = self.results[experiment_id]["A"]
        results_b = self.results[experiment_id]["B"]
        
        if len(results_a) < config.min_sample_size or len(results_b) < config.min_sample_size:
            return {
                "status": "insufficient_data",
                "message": f"需要至少{config.min_sample_size}样本，"
                          f"当前A组:{len(results_a)}, B组:{len(results_b)}"
            }
        
        # 计算各项指标
        analysis = {
            "experiment_id": experiment_id,
            "sample_sizes": {"A": len(results_a), "B": len(results_b)},
            "metrics": {},
            "cost_analysis": {},
            "latency_analysis": {},
            "recommendation": None
        }
        
        # 主要指标对比
        metric_key = config.success_metric
        scores_a = [r["metrics"].get(metric_key, 0) for r in results_a]
        scores_b = [r["metrics"].get(metric_key, 0) for r in results_b]
        
        import numpy as np
        
        analysis["metrics"][metric_key] = {
            "A_mean": np.mean(scores_a),
            "B_mean": np.mean(scores_b),
            "difference": np.mean(scores_b) - np.mean(scores_a),
            "percent_change": (np.mean(scores_b) - np.mean(scores_a)) / np.mean(scores_a) * 100
        }
        
        # 成本分析
        costs_a = [r["cost"] for r in results_a]
        costs_b = [r["cost"] for r in results_b]
        
        analysis["cost_analysis"] = {
            "A_avg_cost": np.mean(costs_a),
            "B_avg_cost": np.mean(costs_b),
            "cost_increase": np.mean(costs_b) - np.mean(costs_a),
            "cost_increase_percent": (np.mean(costs_b) - np.mean(costs_a)) / np.mean(costs_a) * 100
        }
        
        # 延迟分析
        latencies_a = [r["latency_ms"] for r in results_a]
        latencies_b = [r["latency_ms"] for r in results_b]
        
        analysis["latency_analysis"] = {
            "A_p50": np.percentile(latencies_a, 50),
            "A_p99": np.percentile(latencies_a, 99),
            "B_p50": np.percentile(latencies_b, 50),
            "B_p99": np.percentile(latencies_b, 99)
        }
        
        # 生成建议
        analysis["recommendation"] = self._generate_recommendation(analysis)
        
        return analysis
    
    def _generate_recommendation(self, analysis: Dict) -> str:
        """生成实验建议"""
        metrics = analysis["metrics"]
        cost = analysis["cost_analysis"]
        
        # 找到主要指标
        main_metric = list(metrics.keys())[0]
        metric_data = metrics[main_metric]
        
        quality_improvement = metric_data["percent_change"]
        cost_increase = cost["cost_increase_percent"]
        
        # ROI计算
        roi = quality_improvement / max(cost_increase, 0.01)
        
        if quality_improvement > 10 and roi > 2:
            return f"强烈推荐采用B组模型: 质量提升{quality_improvement:.1f}%, 成本仅增加{cost_increase:.1f}%, ROI={roi:.1f}"
        elif quality_improvement > 5 and cost_increase < 20:
            return f"建议采用B组模型: 质量提升{quality_improvement:.1f}%, 成本增加{cost_increase:.1f}%"
        elif quality_improvement > 0:
            return f"B组略有提升({quality_improvement:.1f}%), 可根据预算决定是否切换"
        else:
            return f"建议保持A组: B组质量下降{abs(quality_improvement):.1f}%"

# 多臂老虎机 (MAB) 动态流量分配
class MultiArmedBandit:
    """多臂老虎机 - 动态流量分配"""
    
    def __init__(self, variants: List[str], epsilon: float = 0.1):
        self.variants = variants
        self.epsilon = epsilon  # 探索率
        
        # 每个变体的统计
        self.counts = {v: 0 for v in variants}
        self.rewards = {v: 0.0 for v in variants}
    
    def select_variant(self) -> str:
        """选择变体 (epsilon-greedy)"""
        import random
        
        if random.random() < self.epsilon:
            # 探索: 随机选择
            return random.choice(self.variants)
        else:
            # 利用: 选择平均收益最高的
            avg_rewards = {
                v: self.rewards[v] / max(self.counts[v], 1)
                for v in self.variants
            }
            return max(avg_rewards, key=avg_rewards.get)
    
    def update(self, variant: str, reward: float):
        """更新统计"""
        self.counts[variant] += 1
        self.rewards[variant] += reward
    
    def get_stats(self) -> Dict:
        """获取统计信息"""
        return {
            v: {
                "count": self.counts[v],
                "total_reward": self.rewards[v],
                "avg_reward": self.rewards[v] / max(self.counts[v], 1)
            }
            for v in self.variants
        }
```

```python
    async def optimized_generate(
        self,
        prompt: str,
        user_id: str,
        budget_usd: float = 0.05,
        **kwargs
    ) -> Dict:
        """优化后的生成"""
        
        # 1. Prompt优化
        optimized_prompt = self.prompt_optimizer.compress_prompt(prompt, target_ratio=0.8)
        
        # 2. 选择路由策略

            strategy = "cost"

            strategy = "ab_test"
        else:
            strategy = "smart"
        
        # 3. 生成
        result = await self.gateway.generate(
            optimized_prompt,
            routing_strategy=strategy,
            user_id=user_id,
            max_cost=budget_usd,
            **kwargs
        )
        
        # 4. 记录详细成本
        self.cost_tracker.record(
            model=result["model_used"],
            input_tokens=result.get("input_tokens", 0),
            output_tokens=result.get("output_tokens", 0),
            request_id=result.get("request_id"),
            user_id=user_id,

        )

        # 10%的用户参与A/B测试

        start = end - timedelta(days=days)

        breakdown = self.cost_tracker.get_cost_breakdown(start, end, "model")
        
        # 优化效果
        optimizations = {
            "prompt_compression_savings": self._calculate_compression_savings(),
            "smart_routing_savings": self._calculate_routing_savings(),
            "cache_hit_savings": self._calculate_cache_savings()

            "total_cost": sum(self.cost_tracker.get_daily_cost(
                end - timedelta(days=i)

            "cost_by_model": breakdown,
            "optimization_impact": optimizations,
            "recommendations": self._generate_recommendations(breakdown)
        }

        ]
        if high_cost_requests:
            recommendations.append(
                f"以下模型单次请求成本过高: {', '.join(high_cost_requests)}"
            )
```

## 深度分析

### 成本优化的本质是架构决策

LLM成本不是运维问题，而是架构问题。每一层架构选择都直接影响成本：模型选型决定Token单价基准，Prompt设计决定每次调用的Token消耗量，缓存策略决定重复计算的规避率，路由策略决定高成本模型的调用比例。架构师应当将成本作为一阶约束纳入系统设计——就像在设计分布式系统时考虑延迟一样自然。

### 智能路由：成本优化的核心杠杆

在所有成本优化手段中，智能模型路由的ROI最高（15-30%节省）。核心思想是"让合适的任务找到合适的模型"：简单问答走轻量模型（GPT-4o-mini, $0.15/M），复杂推理走强模型（GPT-4o, $5/M），代码任务走Claude（$3/M），中文任务走Qwen（¥2/M）。实施的关键挑战在于准确评估任务复杂度——过于激进的降级策略会损害用户体验，过于保守则达不到降本效果。

### 成本治理的文化转型

成本优化的技术手段只解决一半问题，另一半是组织文化和流程。需要建立: (1) 成本责任制——每个团队/项目看到自己的Token消耗和账单；(2) 成本SLO——设定每请求成本预算，超支告警；(3) 定期成本评审——每周Review成本报告，识别异常和优化机会。FinOps在AI领域的实践远比传统云计算复杂，因为成本直接与模型输出质量挂钩。
