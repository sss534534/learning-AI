```python
from prometheus_client import Counter, Histogram, Gauge, Info
from typing import Dict, Optional
import time

class LLMMetricsCollector:
    """LLM指标收集器"""
    
    def __init__(self, model_name: str, model_version: str):
        self.model_name = model_name
        self.model_version = model_version
        
        # 请求指标
        self.requests_total = Counter(
            'llm_requests_total',
            'Total LLM requests',
            ['model', 'version', 'status', 'endpoint']
        )
        
        # Token指标
        self.tokens_input = Counter(
            'llm_tokens_input_total',
            'Input tokens count',
            ['model', 'version']
        )
        self.tokens_output = Counter(
            'llm_tokens_output_total',
            'Output tokens count',
            ['model', 'version']
        )
        self.tokens_total = Counter(
            'llm_tokens_total',
            'Total tokens (input + output)',
            ['model', 'version']
        )
        
        # 延迟指标 (分阶段)
        self.latency_prefill = Histogram(
            'llm_latency_prefill_seconds',
            'Time to first token (TTFT)',
            ['model', 'version'],
            buckets=[0.01, 0.025, 0.05, 0.1, 0.25, 0.5, 1.0, 2.5, 5.0, 10.0]
        )
        self.latency_per_token = Histogram(
            'llm_latency_per_token_seconds',
            'Time per output token (TPOT)',
            ['model', 'version'],
            buckets=[0.001, 0.005, 0.01, 0.025, 0.05, 0.1, 0.25]
        )
        self.latency_total = Histogram(
            'llm_latency_total_seconds',
            'Total request latency',
            ['model', 'version'],
            buckets=[0.1, 0.25, 0.5, 1.0, 2.5, 5.0, 10.0, 30.0, 60.0]
        )
        
        # 质量指标
        self.response_length = Histogram(
            'llm_response_length_chars',
            'Response length in characters',
            ['model', 'version'],
            buckets=[100, 250, 500, 1000, 2000, 5000, 10000]
        )
        
        # 成本指标
        self.cost_dollars = Counter(
            'llm_cost_dollars_total',
            'Total cost in USD',
            ['model', 'version', 'cost_type']  # input/output
        )
        
        # 安全指标
        self.safety_violations = Counter(
            'llm_safety_violations_total',
            'Safety policy violations',
            ['model', 'version', 'violation_type']
        )
        
        # 缓存指标
        self.cache_hits = Counter(
            'llm_cache_hits_total',
            'Cache hits',
            ['model', 'version', 'cache_type']  # prompt/prefix/semantic
        )
        self.cache_misses = Counter(
            'llm_cache_misses_total',
            'Cache misses',
            ['model', 'version', 'cache_type']
        )
        
        # 运行时指标
        self.active_requests = Gauge(
            'llm_active_requests',
            'Currently active requests',
            ['model', 'version']
        )
        self.queue_size = Gauge(
            'llm_queue_size',
            'Request queue size',
            ['model', 'version']
        )
        
        # 模型信息
        self.model_info = Info(
            'llm_model',
            'Model metadata'
        )
        self.model_info.info({
            'name': model_name,
            'version': model_version,
            'framework': 'vllm',  # 或 'tgi', 'triton' 等
            'quantization': 'fp16'
        })
    
    def record_request(
        self,
        status: str,
        endpoint: str,
        input_tokens: int,
        output_tokens: int,
        latency_ms: float,
        ttft_ms: Optional[float] = None,
        response_chars: int = 0
    ):
        """记录请求指标"""
        labels = {'model': self.model_name, 'version': self.model_version}
        
        # 请求计数
        self.requests_total.labels(
            model=self.model_name,
            version=self.model_version,
            status=status,
            endpoint=endpoint
        ).inc()
        
        # Token计数
        self.tokens_input.labels(**labels).inc(input_tokens)
        self.tokens_output.labels(**labels).inc(output_tokens)
        self.tokens_total.labels(**labels).inc(input_tokens + output_tokens)
        
        # 延迟
        self.latency_total.labels(**labels).observe(latency_ms / 1000)
        if ttft_ms:
            self.latency_prefill.labels(**labels).observe(ttft_ms / 1000)
            if output_tokens > 0:
                tpot = (latency_ms - ttft_ms) / output_tokens / 1000
                self.latency_per_token.labels(**labels).observe(tpot)
        
        # 响应长度
        self.response_length.labels(**labels).observe(response_chars)
    
    def record_cost(self, input_cost: float, output_cost: float):
        """记录成本"""
        labels = {'model': self.model_name, 'version': self.model_version}
        self.cost_dollars.labels(**labels, cost_type='input').inc(input_cost)
        self.cost_dollars.labels(**labels, cost_type='output').inc(output_cost)
    
    def record_safety_violation(self, violation_type: str):
        """记录安全违规"""
        self.safety_violations.labels(
            model=self.model_name,
            version=self.model_version,
            violation_type=violation_type
        ).inc()
    
    def record_cache(self, hit: bool, cache_type: str = 'prompt'):
        """记录缓存命中/未命中"""
        labels = {'model': self.model_name, 'version': self.model_version}
        if hit:
            self.cache_hits.labels(**labels, cache_type=cache_type).inc()
        else:
            self.cache_misses.labels(**labels, cache_type=cache_type).inc()
    
    def update_active_requests(self, delta: int):
        """更新活跃请求数"""
        self.active_requests.labels(
            model=self.model_name,
            version=self.model_version
        ).set(delta)
    
    @contextmanager
    def measure_request(self, endpoint: str = 'chat'):
        """请求上下文管理器 - 自动测量指标"""
        start_time = time.time()
        self.active_requests.labels(
            model=self.model_name,
            version=self.model_version
        ).inc()
        
        try:
            yield self
            status = 'success'
        except Exception as e:
            status = 'error'
            raise
        finally:
            latency_ms = (time.time() - start_time) * 1000
            self.active_requests.labels(
                model=self.model_name,
                version=self.model_version
            ).dec()
```

```python
from opentelemetry import trace
from opentelemetry.exporter.otlp.proto.grpc.trace_exporter import OTLPSpanExporter
from opentelemetry.sdk.trace import TracerProvider
from opentelemetry.sdk.trace.export import BatchSpanProcessor
from opentelemetry.instrumentation.requests import RequestsInstrumentor

class LLMTracer:
    """LLM追踪器"""
    
    def __init__(self, service_name: str, otlp_endpoint: str):
        # 配置Tracer
        provider = TracerProvider()
        processor = BatchSpanProcessor(
            OTLPSpanExporter(endpoint=otlp_endpoint)
        )
        provider.add_span_processor(processor)
        trace.set_tracer_provider(provider)
        
        self.tracer = trace.get_tracer(service_name)
        
        # 自动 instrument HTTP请求
        RequestsInstrumentor().instrument()
    
    def trace_llm_call(
        self,
        model: str,
        prompt: str,
        parameters: Dict[str, Any]
    ):
        """LLM调用追踪上下文"""
        return self.tracer.start_as_current_span(
            name="llm.generate",
            kind=trace.SpanKind.CLIENT,
            attributes={
                "llm.model": model,
                "llm.prompt_length": len(prompt),
                "llm.temperature": parameters.get("temperature"),
                "llm.max_tokens": parameters.get("max_tokens"),
            }
        )
    
    def trace_rag_pipeline(
        self,
        query: str,
        retriever_type: str
    ):
        """RAG流程追踪"""
        with self.tracer.start_as_current_span("rag.pipeline") as span:
            span.set_attribute("rag.query", query[:100])
            span.set_attribute("rag.retriever", retriever_type)
            
            # 检索阶段
            with self.tracer.start_span("rag.retrieve") as retrieve_span:
                # 执行检索
                documents = self._retrieve(query)
                retrieve_span.set_attribute("rag.documents_count", len(documents))
            
            # 生成阶段
            with self.tracer.start_span("rag.generate") as generate_span:
                # 执行生成
                response = self._generate(query, documents)
                generate_span.set_attribute("llm.response_length", len(response))
            
            return response
    
    def trace_agent_execution(
        self,
        agent_name: str,
        task: str
    ):
        """Agent执行追踪"""
        with self.tracer.start_as_current_span("agent.execute") as span:
            span.set_attribute("agent.name", agent_name)
            span.set_attribute("agent.task", task[:200])
            
            step_count = 0
            while not self._is_task_complete():
                with self.tracer.start_span(f"agent.step.{step_count}") as step_span:
                    # 执行步骤
                    action = self._plan_next_action()
                    step_span.set_attribute("agent.action", action["type"])
                    
                    if action["type"] == "tool_call":
                        with self.tracer.start_span("agent.tool_call") as tool_span:
                            tool_span.set_attribute("tool.name", action["tool"])
                            result = self._call_tool(action)
                            tool_span.set_attribute("tool.success", result["success"])
                    
                    step_count += 1
            
            span.set_attribute("agent.steps", step_count)
```

```python
from langsmith import Client as LangSmithClient
from langfuse import Langfuse
from typing import List, Dict, Any

class LLMObservabilityPlatform:
    """LLM可观测性平台集成"""
    
    def __init__(
        self,
        langsmith_api_key: Optional[str] = None,
        langfuse_public_key: Optional[str] = None,
        langfuse_secret_key: Optional[str] = None
    ):
        self.langsmith = None
        self.langfuse = None
        
        if langsmith_api_key:
            self.langsmith = LangSmithClient(api_key=langsmith_api_key)
        
        if langfuse_public_key and langfuse_secret_key:
            self.langfuse = Langfuse(
                public_key=langfuse_public_key,
                secret_key=langfuse_secret_key
            )
    
    def trace_chat_completion(
        self,
        messages: List[Dict[str, str]],
        model: str,
        response: str,
        metadata: Dict[str, Any] = None
    ):
        """追踪对话完成"""
        # LangSmith追踪
        if self.langsmith:
            self.langsmith.create_run(
                name="chat_completion",
                run_type="llm",
                inputs={"messages": messages},
                outputs={"response": response},
                extra={
                    "model": model,
                    **(metadata or {})
                }
            )
        
        # Langfuse追踪
        if self.langfuse:
            trace = self.langfuse.trace(
                name="chat_completion",
                metadata=metadata
            )
            trace.generation(
                name="llm_call",
                model=model,
                input=messages,
                output=response
            )
    
    def trace_rag(
        self,
        query: str,
        retrieved_docs: List[Dict],
        response: str,
        relevance_scores: List[float]
    ):
        """追踪RAG流程"""
        if self.langsmith:
            self.langsmith.create_run(
                name="rag_pipeline",
                run_type="chain",
                inputs={"query": query},
                outputs={"response": response},
                extra={
                    "retrieved_documents": len(retrieved_docs),
                    "avg_relevance": sum(relevance_scores) / len(relevance_scores) if relevance_scores else 0
                }
            )
        
        if self.langfuse:
            trace = self.langfuse.trace(name="rag_pipeline")
            trace.span(
                name="retrieval",
                input=query,
                output={"documents": retrieved_docs, "scores": relevance_scores}
            )
            trace.generation(
                name="generation",
                input={"query": query, "context": retrieved_docs},
                output=response
            )
    
    def evaluate_response(
        self,
        prompt: str,
        response: str,
        reference: Optional[str] = None
    ) -> Dict[str, float]:
        """评估响应质量"""
        scores = {}
        
        # 使用LangSmith评估器
        if self.langsmith:
            from langsmith.evaluation import evaluate
            
            # 定义评估数据集
            dataset = [{"input": prompt, "output": response, "reference": reference}]
            
            # 运行评估
            results = evaluate(
                dataset,
                evaluators=[
                    "criteria",  # 标准评估
                    "cot_qa",    # 链式思考QA
                ]
            )
            scores["langsmith"] = results
        
        # 使用Langfuse评分
        if self.langfuse:
            # 自定义评分
            scores["langfuse"] = {
                "coherence": self._score_coherence(response),
                "relevance": self._score_relevance(prompt, response),
                "safety": self._score_safety(response)
            }
        
        return scores
    
    def _score_coherence(self, text: str) -> float:
        """评估连贯性 (简化示例)"""
        # 实际应使用专门的评估模型
        sentences = text.split('.')
        return min(1.0, len(sentences) / 5)  # 至少5个句子得满分
    
    def _score_relevance(self, prompt: str, response: str) -> float:
        """评估相关性"""
        # 使用embedding相似度
        # 简化示例
        return 0.85
    
    def _score_safety(self, text: str) -> float:
        """评估安全性"""
        # 使用内容审核API
        # 简化示例
        return 0.95
```

```python
    def add_metric(self, name: str, evaluator: Callable):
        """添加评估指标"""
        self.metrics[name] = evaluator
    
    def evaluate(
        self,
        predictions: List[str],
        references: List[str],
        contexts: List[str] = None
    ) -> Dict[str, EvaluationResult]:
        """执行评估"""
        results = {}
        
        for metric_name, evaluator in self.metrics.items():
            score, details = evaluator(predictions, references, contexts)
            
            # 收集样本
            samples = [
                {
                    "prediction": pred,
                    "reference": ref,
                    "context": ctx,
                    "score": details.get("sample_scores", [])[i] if "sample_scores" in details else None
                }
                for i, (pred, ref, ctx) in enumerate(zip(predictions, references, contexts or []))
            ]
            
            results[metric_name] = EvaluationResult(
                metric_name=metric_name,
                score=score,
                details=details,
                samples=samples
            )
        
        return results

class BuiltInMetrics:
    """内置评估指标"""
    
    @staticmethod
    def bleu_score(predictions: List[str], references: List[str], **kwargs):
        """BLEU分数"""
        from nltk.translate.bleu_score import sentence_bleu, SmoothingFunction
        
        scores = []
        for pred, ref in zip(predictions, references):
            ref_tokens = [ref.split()]
            pred_tokens = pred.split()
            score = sentence_bleu(
                ref_tokens, 
                pred_tokens,
                smoothing_function=SmoothingFunction().method1
            )
            scores.append(score)
        
        return np.mean(scores), {"sample_scores": scores}
    
    @staticmethod
    def rouge_score(predictions: List[str], references: List[str], **kwargs):
        """ROUGE分数"""
        from rouge import Rouge
        
        rouge = Rouge()
        scores = rouge.get_scores(predictions, references, avg=True)
        
        return scores["rouge-l"]["f"], {"full_scores": scores}
    
    @staticmethod
    def semantic_similarity(
        predictions: List[str], 
        references: List[str],
        embedding_model = None
    ):
        """语义相似度"""
        from sentence_transformers import SentenceTransformer
        
        if embedding_model is None:
            embedding_model = SentenceTransformer('all-MiniLM-L6-v2')
        
        pred_embeddings = embedding_model.encode(predictions)
        ref_embeddings = embedding_model.encode(references)
        
        # 计算余弦相似度
        similarities = np.diag(
            np.dot(pred_embeddings, ref_embeddings.T) /
            (np.linalg.norm(pred_embeddings, axis=1) * np.linalg.norm(ref_embeddings, axis=1))
        )
        
        return np.mean(similarities), {"sample_scores": similarities.tolist()}
    
    @staticmethod
    def llm_judge_score(
        predictions: List[str],
        references: List[str],
        contexts: List[str],
        judge_prompt: str = None
    ):
        """LLM作为评判者"""

1分: 完全错误

请只输出数字分数。"""

            # 调用评判模型

            scores.append(score)
        
        return np.mean(scores) / 5.0, {"sample_scores": scores}  # 归一化到0-1

# RAG专用评估
class RAGEvaluator:
    """RAG系统评估器"""
    
    def __init__(self):
        self.metrics = {}
    
    def evaluate_retrieval(
        self,
        queries: List[str],
        retrieved_docs: List[List[str]],
        ground_truth_docs: List[List[str]]
    ) -> Dict[str, float]:
        """评估检索质量"""
        
        # Recall@K
        recalls = []
        for query, retrieved, truth in zip(queries, retrieved_docs, ground_truth_docs):
            truth_set = set(truth)
            retrieved_set = set(retrieved)
            if truth_set:
                recall = len(truth_set & retrieved_set) / len(truth_set)
                recalls.append(recall)
        
        # MRR
        mrrs = []
        for query, retrieved, truth in zip(queries, retrieved_docs, ground_truth_docs):
            for i, doc in enumerate(retrieved, 1):
                if doc in truth:
                    mrrs.append(1.0 / i)
                    break
            else:
                mrrs.append(0)
        
        return {
            "recall@k": np.mean(recalls),
            "mrr": np.mean(mrrs)
        }
    
    def evaluate_generation(
        self,
        queries: List[str],
        responses: List[str],
        contexts: List[List[str]],
        ground_truth_answers: List[str]
    ) -> Dict[str, float]:
        """评估生成质量"""
        
        # 忠实度 (Faithfulness) - 答案是否基于检索内容
        faithfulness_scores = []
        for response, context in zip(responses, contexts):
            score = self._check_faithfulness(response, context)
            faithfulness_scores.append(score)
        
        # 答案相关性
        relevance_scores = []
        for query, response in zip(queries, responses):
            score = self._check_answer_relevance(query, response)
            relevance_scores.append(score)
        
        # 上下文相关性
        context_scores = []
        for query, context in zip(queries, contexts):
            score = self._check_context_relevance(query, context)
            context_scores.append(score)
        
        return {
            "faithfulness": np.mean(faithfulness_scores),
            "answer_relevance": np.mean(relevance_scores),
            "context_relevance": np.mean(context_scores)
        }
    
    def _check_faithfulness(self, response: str, context: List[str]) -> float:
        """检查答案忠实度"""
        # 使用NLI模型或LLM判断
        # 简化：检查关键词重叠
        context_text = " ".join(context).lower()
        response_words = set(response.lower().split())
        context_words = set(context_text.split())
        
        if not response_words:
            return 0.0
        
        overlap = len(response_words & context_words) / len(response_words)
        return min(1.0, overlap * 2)  # 放大信号
    
    def _check_answer_relevance(self, query: str, response: str) -> float:
        """检查答案相关性"""
        # 使用embedding相似度
        return 0.85  # 简化
    
    def _check_context_relevance(self, query: str, context: List[str]) -> float:
        """检查上下文相关性"""
        # 计算查询与每个文档的相关性
        return 0.80  # 简化
```

```python
    def classify_data(
        self,
        content: str,
        metadata: dict = None
    ) -> DataClassification:
        """自动数据分类"""
        
        # PII检测
        pii_patterns = [
            r'\b\d{18}\b',  # 身份证号
            r'\b1[3-9]\d{9}\b',  # 手机号
            r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b',  # 邮箱
        ]
        
        import re
        for pattern in pii_patterns:
            if re.search(pattern, content):
                return DataClassification.PII
        
        # 关键词检测
        sensitive_keywords = ['密码', '密钥', 'token', 'secret', 'confidential']
        if any(kw in content.lower() for kw in sensitive_keywords):
            return DataClassification.CONFIDENTIAL
        
        return DataClassification.INTERNAL

        self,

    ):

            DataClassification.PUBLIC: DataRetentionPolicy.MEDIUM_TERM,
            DataClassification.INTERNAL: DataRetentionPolicy.MEDIUM_TERM,
            DataClassification.CONFIDENTIAL: DataRetentionPolicy.SHORT_TERM,
            DataClassification.RESTRICTED: DataRetentionPolicy.SHORT_TERM,
            DataClassification.PII: DataRetentionPolicy.SHORT_TERM
        }
        
        policy = policy_map.get(classification, DataRetentionPolicy.MEDIUM_TERM)
        
        if policy.value:
            expiry_date = created_at + policy.value
        else:
            expiry_date = None
```

## 总结

### 可观测性最佳实践

1. **指标设计**
   - 分层指标：基础设施、模型服务、业务应用
   - 黄金信号：延迟、流量、错误、饱和度
   - LLM专属：Token消耗、成本、质量评分

2. **日志管理**
   - 结构化日志：便于查询和分析
   - 敏感信息脱敏：保护用户隐私
   - 保留策略：合规与成本的平衡

3. **追踪体系**
   - 分布式追踪：端到端请求链路
   - RAG追踪：检索-生成完整流程
   - Agent追踪：多步骤决策过程

4. **治理合规**
   - 数据分类：自动识别敏感数据
   - 访问控制：最小权限原则
   - 审计追踪：完整的操作记录
   - 模型生命周期：从实验到退役
