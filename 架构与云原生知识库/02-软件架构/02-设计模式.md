### 1.1 单例模式

```python
# 单例模式 - Python实现
class Singleton:
    _instance = None
    
    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

# 使用
s1 = Singleton()
s2 = Singleton()
assert s1 is s2
```

### 1.2 工厂方法

```python
# 工厂方法
class Button(ABC):
    @abstractmethod
    def render(self):
        pass

class WindowsButton(Button):
    def render(self):
        return "Windows按钮"

class MacButton(Button):
    def render(self):
        return "Mac按钮"

class ButtonFactory:
    @staticmethod
    def create_button(os_type):
        if os_type == "windows":
            return WindowsButton()
        elif os_type == "mac":
            return MacButton()
        else:
            raise ValueError("不支持的操作系统")

# 使用
button = ButtonFactory.create_button("windows")
print(button.render())
```

### 1.3 建造者模式

```python
# 建造者模式
class Computer:
    def __init__(self):
        self.cpu = None
        self.memory = None
        self.storage = None
    
    def __str__(self):
        return f"CPU: {self.cpu}, 内存: {self.memory}, 存储: {self.storage}"

class ComputerBuilder:
    def __init__(self):
        self.computer = Computer()
    
    def with_cpu(self, cpu):
        self.computer.cpu = cpu
        return self
    
    def with_memory(self, memory):
        self.computer.memory = memory
        return self
    
    def with_storage(self, storage):
        self.computer.storage = storage
        return self
    
    def build(self):
        return self.computer

# 使用
gaming_computer = ComputerBuilder()\
    .with_cpu("Intel i9")\
    .with_memory("64GB")\
    .with_storage("2TB SSD")\
    .build()

print(gaming_computer)
```

## 2. 结构型模式

### 2.1 适配器模式

```python
# 适配器模式
class LegacyPrinter:
    def print_document(self, content):
        print(f"旧版打印机: {content}")

class ModernPrinter:
    def print(self, content):
        print(f"现代打印机: {content}")

class PrinterAdapter:
    def __init__(self, printer):
        self.printer = printer
    
    def print_document(self, content):
        if hasattr(self.printer, 'print_document'):
            self.printer.print_document(content)
        elif hasattr(self.printer, 'print'):
            self.printer.print(content)

# 使用
legacy = LegacyPrinter()
modern = ModernPrinter()

adapter1 = PrinterAdapter(legacy)
adapter1.print_document("测试文档")

adapter2 = PrinterAdapter(modern)
adapter2.print_document("测试文档")
```

### 2.2 装饰器模式

```python
# 装饰器模式
class Coffee:
    def cost(self):
        return 10
    
    def description(self):
        return "简单咖啡"

class CoffeeDecorator(Coffee):
    def __init__(self, coffee):
        self._coffee = coffee
    
    def cost(self):
        return self._coffee.cost()
    
    def description(self):
        return self._coffee.description()

class Milk(CoffeeDecorator):
    def cost(self):
        return super().cost() + 5
    
    def description(self):
        return super().description() + " + 牛奶"

class Sugar(CoffeeDecorator):
    def cost(self):
        return super().cost() + 2
    
    def description(self):
        return super().description() + " + 糖"

# 使用
coffee = Coffee()
print(f"{coffee.description()}: {coffee.cost()}元")

coffee_with_milk = Milk(coffee)
print(f"{coffee_with_milk.description()}: {coffee_with_milk.cost()}元")

coffee_with_all = Sugar(Milk(coffee))
print(f"{coffee_with_all.description()}: {coffee_with_all.cost()}元")
```

## 3. 行为型模式

### 3.1 观察者模式

```python
# 观察者模式
class Subject:
    def __init__(self):
        self._observers = []
    
    def attach(self, observer):
        if observer not in self._observers:
            self._observers.append(observer)
    
    def detach(self, observer):
        self._observers.remove(observer)
    
    def notify(self, message):
        for observer in self._observers:
            observer.update(message)

class Observer(ABC):
    @abstractmethod
    def update(self, message):
        pass

class NewsReader(Observer):
    def __init__(self, name):
        self.name = name
    
    def update(self, message):
        print(f"{self.name} 收到新闻: {message}")

class NewsAgency(Subject):
    def publish_news(self, news):
        print(f"发布新闻: {news}")
        self.notify(news)

# 使用
agency = NewsAgency()
reader1 = NewsReader("张三")
reader2 = NewsReader("李四")

agency.attach(reader1)
agency.attach(reader2)

agency.publish_news("K8s 1.30发布了！")

agency.detach(reader1)
agency.publish_news("Docker 26.0更新了！")
```

### 3.2 策略模式

```python
# 策略模式
class SortStrategy(ABC):
    @abstractmethod
    def sort(self, data):
        pass

class QuickSort(SortStrategy):
    def sort(self, data):
        return sorted(data)

class BubbleSort(SortStrategy):
    def sort(self, data):
        n = len(data)
        for i in range(n):
            for j in range(0, n-i-1):
                if data[j] > data[j+1]:
                    data[j], data[j+1] = data[j+1], data[j]
        return data

class SortContext:
    def __init__(self, strategy):
        self._strategy = strategy
    
    def set_strategy(self, strategy):
        self._strategy = strategy
    
    def sort_data(self, data):
        return self._strategy.sort(data.copy())

# 使用
data = [3, 1, 4, 1, 5, 9, 2, 6]
context = SortContext(QuickSort())
print("快速排序结果:", context.sort_data(data))

context.set_strategy(BubbleSort())
print("冒泡排序结果:", context.sort_data(data))
```
