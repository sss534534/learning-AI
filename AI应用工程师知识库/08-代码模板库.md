### 1.1 基础RAG应用

```python
"""
基础RAG应用模板
功能：文档问答系统
"""

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import Chroma
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.chains import RetrievalQA
from langchain.document_loaders import PyPDFLoader
from langchain.prompts import PromptTemplate
import os

class BasicRAGApplication:
    """基础RAG应用类"""
    
    def __init__(self, api_key: str, model: str = "gpt-4o"):
        self.llm = ChatOpenAI(
            model=model,
            temperature=0,
            api_key=api_key
        )
        self.embeddings = OpenAIEmbeddings(api_key=api_key)
        self.vectorstore = None
        self.qa_chain = None
    
    def load_documents(self, file_path: str):
        """加载文档"""
        loader = PyPDFLoader(file_path)
        documents = loader.load()
        
        # 分块
        text_splitter = RecursiveCharacterTextSplitter(
            chunk_size=1000,
            chunk_overlap=200
        )
        chunks = text_splitter.split_documents(documents)
        
        # 向量化存储
        self.vectorstore = Chroma.from_documents(
            documents=chunks,
            embedding=self.embeddings
        )
        
        return len(chunks)
    
    def setup_qa_chain(self):
        """设置问答链"""
        retriever = self.vectorstore.as_retriever(
            search_kwargs={"k": 3}
        )
        
        prompt = PromptTemplate(
            template="""基于以下上下文回答问题。如果上下文中没有相关信息，
            请说明你不知道。

            上下文：
            {context}

            问题：{question}

            回答：""",
            input_variables=["context", "question"]
        )
        
        self.qa_chain = RetrievalQA.from_chain_type(
            llm=self.llm,
            retriever=retriever,
            chain_type="stuff",
            chain_type_kwargs={"prompt": prompt},
            return_source_documents=True
        )
    
    def query(self, question: str) -> dict:
        """查询"""
        if not self.qa_chain:
            raise ValueError("请先调用 load_documents 和 setup_qa_chain")
        
        result = self.qa_chain.invoke({"query": question})
        
        return {
            "answer": result["result"],
            "sources": [doc.metadata for doc in result["source_documents"]]
        }


# 使用示例
if __name__ == "__main__":
    rag = BasicRAGApplication(
        api_key="your-api-key",
        model="gpt-4o"
    )
    
    # 加载文档
    chunks_count = rag.load_documents("document.pdf")
    print(f"已加载 {chunks_count} 个文本块")
    
    # 设置问答链
    rag.setup_qa_chain()
    
    # 查询
    result = rag.query("文档的主要内容是什么？")
    print(f"答案：{result['answer']}")
```

### 1.2 高级RAG应用（带重排序）

```python
"""
高级RAG应用模板
功能：带重排序和多路召回的RAG系统
"""

from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from langchain_community.vectorstores import Chroma
from langchain_community.retrievers import BM25Retriever
from langchain.retrievers import EnsembleRetriever, ContextualCompressionRetriever
from langchain_cohere import CohereRerank
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain.chains import RetrievalQA
from langchain.prompts import PromptTemplate
from langchain.schema import Document

class AdvancedRAGApplication:
    """高级RAG应用类"""
    
    def __init__(self, api_key: str):
        self.llm = ChatOpenAI(model="gpt-4o", api_key=api_key)
        self.embeddings = OpenAIEmbeddings(api_key=api_key)
        self.vectorstore = None
        self.qa_chain = None
    
    def load_documents(self, documents: list[str], metadata: dict = None):
        """批量加载文档"""
        all_chunks = []
        
        for doc_text in documents:
            text_splitter = RecursiveCharacterTextSplitter(
                chunk_size=800,
                chunk_overlap=100
            )
            chunks = text_splitter.split_text(doc_text)
            
            for i, chunk in enumerate(chunks):
                chunk_metadata = {
                    "chunk_index": i,
                    "total_chunks": len(chunks)
                }
                if metadata:
                    chunk_metadata.update(metadata)
                
                all_chunks.append(Document(
                    page_content=chunk,
                    metadata=chunk_metadata
                ))
        
        self.vectorstore = Chroma.from_documents(
            documents=all_chunks,
            embedding=self.embeddings
        )
        
        return len(all_chunks)
    
    def setup_advanced_retriever(self, cohere_api_key: str):
        """设置高级检索器（多路召回+重排序）"""
        
        # 向量检索
        vector_retriever = self.vectorstore.as_retriever(
            search_kwargs={"k": 10}
        )
        
        # 关键词检索
        texts = [doc.page_content for doc in self.vectorstore.get()["documents"]]
        bm25_retriever = BM25Retriever.from_texts(texts)
        bm25_retriever.k = 10
        
        # 多路召回
        ensemble_retriever = EnsembleRetriever(
            retrievers=[vector_retriever, bm25_retriever],
            weights=[0.6, 0.4]
        )
        
        # 重排序
        compressor = CohereRerank(
            api_key=cohere_api_key,
            top_n=5
        )
        
        # 最终检索器
        self.retriever = ContextualCompressionRetriever(
            base_compressor=compressor,
            base_retriever=ensemble_retriever
        )
    
    def setup_qa_chain(self):
        """设置问答链"""
        prompt = PromptTemplate(
            template="""你是一个专业的问答助手。基于以下上下文信息，
            用清晰、简洁的方式回答问题。

            上下文：
            {context}

            问题：{question}

            要求：
            1. 如果上下文中没有相关信息，诚实回答"我不知道"
            2. 回答要基于上下文中的事实
            3. 如果有多个相关信息，综合回答

            回答：""",
            input_variables=["context", "question"]
        )
        
        self.qa_chain = RetrievalQA.from_chain_type(
            llm=self.llm,
            retriever=self.retriever,
            chain_type="refine",
            chain_type_kwargs={"prompt": prompt},
            return_source_documents=True
        )
    
    def query(self, question: str) -> dict:
        """查询"""
        result = self.qa_chain.invoke({"query": question})
        
        return {
            "answer": result["result"],
            "sources": result["source_documents"]
        }
```

```python
"""
ReAct Agent模板
功能：带工具调用的智能助手
"""

from langchain_openai import ChatOpenAI
from langchain.agents import create_react_agent, AgentExecutor
from langchain.prompts import PromptTemplate
from langchain.tools import tool
from typing import List
import json

# 定义工具
@tool
def search_knowledge_base(query: str) -> str:
    """搜索知识库获取相关信息
    
    Args:
        query: 搜索关键词
    
    Returns:
        搜索结果列表
    """
    # 这里连接实际的知识库
    knowledge_base = {
        "python": "Python是一种高级编程语言...",
        "java": "Java是一种面向对象的编程语言...",
        "llm": "大语言模型是..."
    }
    
    results = []
    for key, value in knowledge_base.items():
        if query.lower() in key.lower() or query.lower() in value.lower():
            results.append(f"{key}: {value}")
    
    return "\n".join(results) if results else "未找到相关信息"

@tool
def calculator(expression: str) -> str:
    """执行数学计算
    
    Args:
        expression: 数学表达式，如 "2 + 3 * 4"
    
    Returns:
        计算结果
    """
    try:
        # 安全计算（仅允许基本运算）
        allowed = set("0123456789+-*/.() ")
        if all(c in allowed for c in expression):
            result = eval(expression)
            return str(result)
        return "错误：表达式包含非法字符"
    except Exception as e:
        return f"计算错误：{str(e)}"

@tool
def get_current_time() -> str:
    """获取当前时间"""
    from datetime import datetime
    return datetime.now().strftime("%Y-%m-%d %H:%M:%S")

class ReActAgentApplication:
    """ReAct Agent应用类"""
    
    def __init__(self, api_key: str):
        self.llm = ChatOpenAI(model="gpt-4o", temperature=0, api_key=api_key)
        self.tools = [search_knowledge_base, calculator, get_current_time]
        self.agent = None
        self.agent_executor = None
    
    def setup_agent(self):
        """设置Agent"""
        prompt = PromptTemplate.from_template("""你是一个智能助手，可以通过工具来完成任务。

Available tools:
{tools}

使用方法：
当你需要使用工具时，按照以下格式：
```
Action: 工具名称
Action Input: 工具输入参数
```

Question: {input}

让我一步步思考：
{agent_scratchpad}""")
        
        self.agent = create_react_agent(
            llm=self.llm,
            tools=self.tools,
            prompt=prompt
        )
        
        self.agent_executor = AgentExecutor(
            agent=self.agent,
            tools=self.tools,
            max_iterations=10,
            verbose=True,
            handle_parsing_errors=True
        )
    
    def run(self, task: str) -> str:
        """运行Agent"""
        if not self.agent_executor:
            raise ValueError("请先调用 setup_agent")
        
        result = self.agent_executor.invoke({"input": task})
        return result["output"]
```

### 2.2 LangGraph状态机Agent模板

```python
"""
LangGraph状态机Agent模板
功能：复杂状态管理和流程控制的Agent
"""

from langchain_openai import ChatOpenAI
from langchain.tools import tool
from langgraph.graph import StateGraph, END
from typing import TypedDict, Annotated
import operator

# 定义状态
class AgentState(TypedDict):
    messages: Annotated[list, operator.add]
    current_task: str
    task_result: str
    next_action: str

# 定义工具
@tool
def search_web(query: str) -> str:
    """搜索网络"""
    return f"关于 '{query}' 的搜索结果..."

@tool
def read_file(filepath: str) -> str:
    """读取文件"""
    try:
        with open(filepath, 'r', encoding='utf-8') as f:
            return f.read()
    except:
        return f"无法读取文件: {filepath}"

@tool
def write_file(filepath: str, content: str) -> str:
    """写入文件"""
    try:
        with open(filepath, 'w', encoding='utf-8') as f:
            f.write(content)
        return f"成功写入文件: {filepath}"
    except:
        return f"无法写入文件: {filepath}"

class StatefulAgentApplication:
    """状态机Agent应用类"""
    
    def __init__(self, api_key: str):
        self.llm = ChatOpenAI(model="gpt-4o", api_key=api_key)
        self.tools = [search_web, read_file, write_file]
        self.graph = None
    
    def should_continue(self, state: AgentState) -> str:
        """判断是否继续"""
        return "act" if state.get("next_action") == "continue" else "end"
    
    def plan_node(self, state: AgentState) -> dict:
        """规划节点"""
        last_message = state["messages"][-1]["content"]
        
        response = self.llm.invoke(
            f"""分析以下任务，决定下一步操作：
            
            任务：{last_message}
            
            可用操作：
            1. search - 需要搜索网络
            2. read - 需要读取文件
            3. write - 需要写入文件
            4. respond - 任务完成，需要回答用户
            
            返回JSON格式：
            {{"action": "操作名称", "content": "操作内容"}}
            """
        )
        
        # 解析响应
        import json
        try:
            action_data = json.loads(response.content)
        except:
            action_data = {"action": "respond", "content": response.content}
        
        return {
            "current_task": action_data.get("action", "respond"),
            "task_result": action_data.get("content", ""),
            "next_action": "continue" if action_data["action"] != "respond" else "end"
        }
    
    def act_node(self, state: AgentState) -> dict:
        """执行节点"""
        task = state["current_task"]
        task_content = state["task_result"]
        
        # 根据任务类型执行
        if task == "search":
            result = search_web.invoke(task_content)
        elif task == "read":
            result = read_file.invoke(task_content)
        elif task == "write":
            result = write_file.invoke(task_content)
        else:
            result = "未知任务"
        
        return {
            "messages": [{"role": "assistant", "content": result}]
        }
    
    def respond_node(self, state: AgentState) -> dict:
        """响应节点"""
        task_result = state["task_result"]
        
        response = self.llm.invoke(
            f"基于以下结果，用友好的方式回答用户：\n\n{task_result}"
        )
        
        return {
            "messages": [{"role": "assistant", "content": response.content}]
        }
    
    def setup_graph(self):
        """设置状态图"""
        workflow = StateGraph(AgentState)
        
        # 添加节点
        workflow.add_node("plan", self.plan_node)
        workflow.add_node("act", self.act_node)
        workflow.add_node("respond", self.respond_node)
        
        # 添加边
        workflow.add_edge("plan", "act")
        workflow.add_conditional_edges(
            "act",
            self.should_continue,
            {
                "continue": "plan",
                "end": END
            }
        )
        workflow.add_edge("respond", END)
        
        # 设置入口
        workflow.set_entry_point("plan")
        
        self.graph = workflow.compile()
    
    def run(self, task: str) -> str:
        """运行Agent"""
        if not self.graph:
            raise ValueError("请先调用 setup_graph")
        
        result = self.graph.invoke({
            "messages": [{"role": "user", "content": task}],
            "current_task": "",
            "task_result": "",
            "next_action": ""
        })
        
        return result["messages"][-1]["content"]
```

### 3.1 多轮对话模板

```python
"""
多轮对话管理模板
功能：带会话历史的智能对话
"""

from langchain_openai import ChatOpenAI
from langchain.memory import ConversationBufferMemory
from langchain.chains import LLMChain
from langchain.prompts import PromptTemplate
from typing import Optional
import json
from datetime import datetime

class ConversationManager:
    """对话管理器"""
    
    def __init__(self, api_key: str):
        self.llm = ChatOpenAI(model="gpt-4o", temperature=0.7, api_key=api_key)
        self.conversations = {}  # session_id -> memory
        self.conversation_limit = 20  # 最大历史消息数
    
    def _create_prompt(self) -> PromptTemplate:
        """创建Prompt模板"""
        template = """你是一个专业的AI助手。请根据对话历史回答用户的问题。

对话历史：
{chat_history}

当前用户：{user_input}

AI助手："""
        return PromptTemplate(
            template=template,
            input_variables=["chat_history", "user_input"]
        )
    
    def create_session(self, session_id: Optional[str] = None) -> str:
        """创建新会话"""
        if session_id is None:
            session_id = f"session_{datetime.now().timestamp()}"
        
        if session_id not in self.conversations:
            self.conversations[session_id] = ConversationBufferMemory(
                memory_key="chat_history",
                return_messages=True,
                output_key="response"
            )
        
        return session_id
    
    def chat(self, session_id: str, user_input: str) -> str:
        """发送消息"""
        if session_id not in self.conversations:
            self.create_session(session_id)
        
        memory = self.conversations[session_id]
        chain = LLMChain(
            llm=self.llm,
            prompt=self._create_prompt(),
            memory=memory
        )
        
        # 执行对话
        response = chain.invoke({
            "user_input": user_input
        })
        
        # 检查并限制历史长度
        self._trim_history_if_needed(session_id)
        
        return response["response"]
    
    def _trim_history_if_needed(self, session_id: str):
        """修剪过长的历史"""
        memory = self.conversations[session_id]
        messages = memory.chat_memory.messages
        
        if len(messages) > self.conversation_limit * 2:
            # 保留最近的消息
            recent_messages = messages[-self.conversation_limit * 2:]
            memory.chat_memory.messages = recent_messages
    
    def get_history(self, session_id: str) -> list:
        """获取对话历史"""
        if session_id not in self.conversations:
            return []
        
        messages = self.conversations[session_id].chat_memory.messages
        return [
            {
                "role": "user" if i % 2 == 0 else "assistant",
                "content": msg.content
            }
            for i, msg in enumerate(messages)
        ]
    
    def clear_session(self, session_id: str):
        """清除会话"""
        if session_id in self.conversations:
            self.conversations[session_id].clear()
    
    def delete_session(self, session_id: str):
        """删除会话"""
        if session_id in self.conversations:
            del self.conversations[session_id]
```

```python
"""
流式对话模板
功能：实时流式响应
"""

from langchain_openai import ChatOpenAI
from fastapi import FastAPI, Request
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
from typing import Optional
import asyncio

app = FastAPI()

# 全局LLM实例
llm = ChatOpenAI(model="gpt-4o", api_key="your-api-key")

class ChatRequest(BaseModel):
    session_id: str
    message: str
    stream: bool = True

@app.post("/chat")
async def chat(request: ChatRequest):
    """流式对话接口"""
    
    if request.stream:
        return StreamingResponse(
            generate_response(request.session_id, request.message),
            media_type="text/event-stream"
        )
    else:
        # 非流式响应
        response = await llm.ainvoke(request.message)
        return {"response": response.content}

async def generate_response(session_id: str, message: str):
    """生成流式响应"""
    try:
        # 发送开始消息
        yield f"data: {json.dumps({'type': 'start', 'session_id': session_id})}\n\n"
        
        # 流式生成
        async for chunk in llm.astream(message):
            if chunk.content:
                yield f"data: {json.dumps({'type': 'chunk', 'content': chunk.content})}\n\n"
        
        # 发送完成消息
        yield f"data: {json.dumps({'type': 'end', 'session_id': session_id})}\n\n"
        
    except Exception as e:
        yield f"data: {json.dumps({'type': 'error', 'error': str(e)})}\n\n"

@app.get("/health")
async def health():
    return {"status": "healthy"}
```

```python
"""

from fastapi import FastAPI, HTTPException, UploadFile, File
from fastapi.middleware.cors import CORSMiddleware

    CORSMiddleware,
    allow_origins=["*"],
    allow_credentials=True,
    allow_methods=["*"],
    allow_headers=["*"],
)

# 请求模型
class QueryRequest(BaseModel):
    question: str
    session_id: Optional[str] = None
    top_k: int = 3

class QueryResponse(BaseModel):
    answer: str
    sources: List[dict]

# 全局RAG实例（实际应该依赖注入）
rag_app = None

@app.on_event("startup")
async def startup():
    """启动时初始化"""
    global rag_app
    # 初始化RAG应用
    from your_module import AdvancedRAGApplication
    rag_app = AdvancedRAGApplication(api_key=os.getenv("OPENAI_API_KEY"))

@app.post("/query", response_model=QueryResponse)
async def query(request: QueryRequest):
    """问答接口"""
    try:
        result = rag_app.query(request.question)
        return QueryResponse(
            answer=result["answer"],
            sources=[{"content": doc.page_content, "metadata": doc.metadata} 
                     for doc in result["sources"]]
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/upload")
async def upload_document(file: UploadFile = File(...)):
    """上传文档"""
    try:
        # 保存临时文件
        with tempfile.NamedTemporaryFile(delete=False, suffix=file.filename) as tmp:
            content = await file.read()
            tmp.write(content)
            tmp_path = tmp.name
        
        # 处理文档
        chunks = rag_app.load_documents([tmp_path], {"filename": file.filename})
        
        # 清理临时文件
        os.unlink(tmp_path)
        
        return {"status": "success", "chunks": chunks}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.get("/health")
async def health():
    return {"status": "healthy"}

if __name__ == "__main__":
    uvicorn.run(app, host="0.0.0.0", port=8000)
```

```python
"""
PostgreSQL + pgvector 集成模板
"""

from langchain_openai import OpenAIEmbeddings
from langchain_community.vectorstores.pgvector import PGVector
from sqlalchemy import create_engine
from sqlalchemy.orm import sessionmaker
import os

class VectorDatabaseManager:
    """向量数据库管理器"""
    
    def __init__(self, connection_string: str):
        self.connection_string = connection_string
        self.embeddings = OpenAIEmbeddings(
            model="text-embedding-3-large"
        )
    
    def get_vectorstore(self, table_name: str = "document_vectors"):
        """获取向量存储"""
        return PGVector.from_documents(
            embedding=self.embeddings,
            connection_string=self.connection_string,
            table_name=table_name,
            pre_delete_table=False  # 不删除已有数据
        )
    
    def similarity_search(self, query: str, k: int = 5):
        """相似性搜索"""
        vectorstore = self.get_vectorstore()
        docs = vectorstore.similarity_search(query, k=k)
        return docs
    
    def similarity_search_with_score(self, query: str, k: int = 5):
        """带分数的相似性搜索"""
        vectorstore = self.get_vectorstore()
        docs_and_scores = vectorstore.similarity_search_with_score(query, k=k)
        return docs_and_scores
    
    def add_documents(self, documents: list):
        """添加文档"""
        vectorstore = self.get_vectorstore()
        vectorstore.add_documents(documents)
```

```python
"""
Redis缓存集成模板
"""

from langchain.cache import RedisCache
from langchain.globals import set_llm_cache
from langchain_openai import ChatOpenAI
import redis
import hashlib
import json

class SemanticCache:
    """语义缓存"""
    
    def __init__(self, redis_url: str = "redis://localhost:6379"):
        self.redis_client = redis.from_url(redis_url)
        self.embeddings = OpenAIEmbeddings()
        self.ttl = 3600  # 缓存有效期：1小时
    
    def _get_cache_key(self, text: str) -> str:
        """生成缓存键"""
        embedding = self.embeddings.embed_query(text)
        # 使用向量前128维作为键
        key_vector = embedding[:128]
        return f"semantic_cache:{hashlib.md5(str(key_vector).encode()).hexdigest()}"
    
    def get(self, query: str) -> str:
        """获取缓存"""
        cache_key = self._get_cache_key(query)
        cached = self.redis_client.get(cache_key)
        if cached:
            return json.loads(cached)
        return None
    
    def set(self, query: str, response: str):
        """设置缓存"""
        cache_key = self._get_cache_key(query)
        self.redis_client.setex(
            cache_key,
            self.ttl,
            json.dumps(response)
        )
    
    def clear(self):
        """清除所有缓存"""
        for key in self.redis_client.scan_iter("semantic_cache:*"):
            self.redis_client.delete(key)


class LLMCache:
    """LLM响应缓存"""
    
    def __init__(self, redis_url: str = "redis://localhost:6379"):
        # 设置LangChain缓存
        redis_cache = RedisCache.from_url(redis_url)
        set_llm_cache(redis_cache)
        
        self.llm = ChatOpenAI(model="gpt-4o")
    
    def cached_chat(self, messages: list) -> str:
        """带缓存的聊天"""
        response = self.llm.invoke(messages)
        return response.content
```

```python
功能：快速创建新的 MCP Server

from mcp.server import Server, NotificationOptions
from mcp.server.models import InitializationOptions
import mcp.server.stdio
import mcp.types as types
from typing import Any

class MCPServer:
    """MCP Server 基类"""

    def __init__(self, name: str, version: str = "1.0.0"):
        self.server = Server(name)
        self.name = name
        self.version = version
        self._register_handlers()

    def _register_handlers(self):
        @self.server.list_tools()
        async def list_tools() -> list[types.Tool]:
            return self.get_tools()

        @self.server.call_tool()
        async def call_tool(
            name: str, arguments: dict[str, Any]
        ) -> list[types.TextContent]:
            return await self.execute_tool(name, arguments)

    def get_tools(self) -> list[types.Tool]:
        """子类实现：声明工具列表"""
        raise NotImplementedError

    async def execute_tool(
        self, name: str, arguments: dict[str, Any]
    ) -> list[types.TextContent]:
        """子类实现：执行工具"""
        raise NotImplementedError

    async def run(self):
        async with mcp.server.stdio.stdio_server() as (read, write):
            await self.server.run(
                read, write,
                InitializationOptions(
                    server_name=self.name,
                    server_version=self.version,
                    capabilities=self.server.get_capabilities(
                        notification_options=NotificationOptions(),
                        experimental_capabilities={},
                    ),
                ),
            )
```

```python
"""
OpenAI Agents SDK 多 Agent 工作流模板

from agents import Agent, Runner, function_tool
from dataclasses import dataclass

@dataclass
class WorkflowContext:
    """跨 Agent 共享上下文"""
    topic: str
    research_notes: str = ""
    draft: str = ""
    final_output: str = ""

@function_tool
def save_context(key: str, value: str) -> str:
    """保存工作流上下文"""
    # 实际场景使用 Redis/数据库
    print(f"[Context] {key}: {value}")
    return f"已保存: {key}"

async def run_workflow(topic: str) -> str:
    """执行完整的工作流"""

    researcher = Agent(
        name="Researcher",
        instructions="搜索技术资料并整理要点",
        tools=[save_context]
    )

    writer = Agent(
        name="Writer",
        instructions="将技术要点组织为文档",
        tools=[save_context]
    )

    # 阶段 1
    research = await Runner.run(
        researcher,
        f"研究 {topic} 的核心概念和最新进展"
    )

    # 阶段 2
    final = await Runner.run(
        writer,
        f"基于以下内容撰写技术文档：\n{research.final_output}"
    )

    return final.final_output
```
