# AI应用架构设计

> 生产级AI应用系统架构指南

## 1. AI应用架构模式

### 1.1 三层架构

```
┌─────────────────────────────────────────────────┐
│                   接入层                          │
│   Web/APP/小程序/API网关/消息队列                │
└──────────────────────┬──────────────────────────┘
                       ↓
┌─────────────────────────────────────────────────┐
│                   业务层                          │
│   对话管理 | 任务编排 | 业务逻辑 | 权限控制        │
└──────────────────────┬──────────────────────────┘
                       ↓
┌─────────────────────────────────────────────────┐
│                   AI服务层                        │
│   LLM服务 | RAG服务 | Agent服务 | Embedding服务   │
└──────────────────────┬──────────────────────────┘
                       ↓
┌─────────────────────────────────────────────────┐
│                   基础设施层                       │
│   向量数据库 | 缓存 | 消息队列 | 对象存储          │
└─────────────────────────────────────────────────┘
```

### 1.2 常见架构模式

| 模式 | 适用场景 | 复杂度 |
|------|----------|--------|
| **直接调用** | 简单问答、翻译 | 低 |
| **RAG增强** | 知识库问答 | 中 |
| **Agent系统** | 复杂任务执行 | 高 |
| **多Agent协作** | 业务流程自动化 | 很高 |

```java
@Service
public class ChatService {
    
    private final ChatClient chatClient;
    private final ChatMemory chatMemory;
    
    public ChatService(ChatClient.Builder builder, ChatMemory chatMemory) {
        this.chatClient = builder
            .defaultSystem("你是一个专业的AI助手")
            .defaultAdvisors(new MessageChatMemoryAdvisor(chatMemory))
            .build();
        this.chatMemory = chatMemory;
    }
    
    public String chat(String sessionId, String message) {
        return chatClient.prompt()
            .user(message)
            .advisors(a -> a.param(CHAT_MEMORY_CONVERSATION_ID_KEY, sessionId))
            .call()
            .content();
    }
    
    public Flux<String> chatStream(String sessionId, String message) {
        return chatClient.prompt()
            .user(message)
            .advisors(a -> a.param(CHAT_MEMORY_CONVERSATION_ID_KEY, sessionId))
            .stream()
            .content();
    }
}
```

```python
from fastapi import FastAPI, HTTPException
from fastapi.responses import StreamingResponse
from pydantic import BaseModel
import asyncio

app = FastAPI(title="AI Application API")

class ChatRequest(BaseModel):
    session_id: str
    message: str
    stream: bool = False

class ChatResponse(BaseModel):
    session_id: str
    answer: str
    tokens_used: int

@app.post("/api/chat", response_model=ChatResponse)
async def chat(request: ChatRequest):
    try:
        service = ChatService()
        
        if request.stream:
            return StreamingResponse(
                service.chat_stream(request.session_id, request.message),
                media_type="text/event-stream"
            )
        
        answer, tokens = await service.chat(request.session_id, request.message)
        return ChatResponse(
            session_id=request.session_id,
            answer=answer,
            tokens_used=tokens
        )
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
```

### 3.2 文档管理服务

```python
class DocumentManagementService:
    """文档管理服务"""
    
    def __init__(self, vectorstore, embeddings, text_splitter):
        self.vectorstore = vectorstore
        self.embeddings = embeddings
        self.text_splitter = text_splitter
    
    async def upload_document(self, file_path: str, metadata: dict):
        """上传并索引文档"""
        # 1. 加载文档
        documents = self._load_document(file_path)
        
        # 2. 添加元数据
        for doc in documents:
            doc.metadata.update({
                "doc_id": str(uuid.uuid4()),
                "source": os.path.basename(file_path),
                "upload_time": datetime.now().isoformat(),
                **metadata
            })
        
        # 3. 分块
        chunks = self.text_splitter.split_documents(documents)
        
        # 4. 向量化存储
        self.vectorstore.add_documents(chunks)
        
        return {"status": "success", "chunks": len(chunks)}
    
    async def delete_document(self, source: str):
        """删除文档（按来源）"""
        ids = self.vectorstore.get_ids_by_metadata({"source": source})
        self.vectorstore.delete(ids)
    
    async def update_document(self, file_path: str, metadata: dict):
        """更新文档（先删后加）"""
        source = os.path.basename(file_path)
        await self.delete_document(source)
        await self.upload_document(file_path, metadata)
```

## 4. 对话系统架构

### 4.1 会话管理

**会话状态机：**
```
会话生命周期：
创建 → 活跃 → 空闲 → 过期 → 归档

状态转换：
- 用户发消息 → 活跃
- 超过30分钟无消息 → 空闲
- 超过24小时 → 过期
- 用户主动清除 → 归档
```

**会话存储设计：**
```python
class SessionManager:
    """会话管理器"""
    
    def __init__(self, redis_client):
        self.redis = redis_client
        self.ttl = 86400 * 7  # 7天过期
    
    async def create_session(self, user_id: str) -> str:
        session_id = str(uuid.uuid4())
        await self.redis.hset(f"session:{session_id}", mapping={
            "user_id": user_id,
            "created_at": datetime.now().isoformat(),
            "message_count": 0,
            "total_tokens": 0
        })
        await self.redis.expire(f"session:{session_id}", self.ttl)
        return session_id
    
    async def add_message(self, session_id: str, role: str, content: str):
        key = f"session:{session_id}:messages"
        await self.redis.rpush(key, json.dumps({
            "role": role,
            "content": content,
            "timestamp": datetime.now().isoformat()
        }))
        # 更新统计
        await self.redis.hincrby(f"session:{session_id}", "message_count", 1)
    
    async def get_history(self, session_id: str, limit: int = 20):
        key = f"session:{session_id}:messages"
        messages = await self.redis.lrange(key, -limit * 2, -1)
        return [json.loads(m) for m in messages]
```

### 4.2 多轮对话优化

**上下文窗口管理策略：**
```python
class ContextWindowManager:
    """上下文窗口管理"""
    
    def __init__(self, max_tokens=120000):
        self.max_tokens = max_tokens
    
    def manage_context(self, messages, system_prompt=""):
        """管理上下文窗口"""
        # 1. 始终保留system prompt
        result = []
        if system_prompt:
            result.append({"role": "system", "content": system_prompt})
        
        # 2. 从最新消息开始，向前添加
        total_tokens = 0
        for msg in reversed(messages):
            msg_tokens = count_tokens(msg["content"])
            if total_tokens + msg_tokens > self.max_tokens:
                break
            result.insert(1, msg)  # 插入到system之后
            total_tokens += msg_tokens
        
        # 3. 如果被裁剪，添加摘要
        if len(messages) > len(result) - 1:
            summary = self._summarize_old_messages(messages[:len(messages) - len(result) + 1])
            result.insert(1, {
                "role": "system",
                "content": f"以下是之前对话的摘要：{summary}"
            })
        
        return result
```

```python
class LLMServiceWithFallback:
    """带降级的LLM服务"""
    
    def __init__(self):
        self.primary = "gpt-4o"
        self.fallback = "gpt-4o-mini"
        self.cache = SemanticCache()
    
    async def chat(self, messages, **kwargs):
        # 1. 检查缓存
        cached = self.cache.get(messages)
        if cached:
            return cached
        
        # 2. 尝试主模型
        try:
            result = await self._call_llm(self.primary, messages, **kwargs)
            self.cache.set(messages, result)
            return result
        except RateLimitError:
            # 降级到小模型
            return await self._call_llm(self.fallback, messages, **kwargs)
        except TimeoutError:
            return {"error": "服务繁忙，请稍后重试"}
```

```python
import logging
from langsmith import traceable

logger = logging.getLogger(__name__)

@traceable(name="rag_query", tags=["production"])
async def rag_query(session_id: str, question: str):
    start_time = time.time()
    
    logger.info(f"[RAG] session={session_id}, query={question[:50]}")
    
    try:
        # 检索
        retrieval_start = time.time()
        docs = retriever.invoke(question)
        retrieval_time = time.time() - retrieval_start
        
        # 生成
        gen_start = time.time()
        answer = generator.invoke({"context": docs, "question": question})
        gen_time = time.time() - gen_start
        
        total_time = time.time() - start_time
        
        logger.info(
            f"[RAG] session={session_id}, "
            f"retrieval={retrieval_time:.2f}s, "
            f"generation={gen_time:.2f}s, "
            f"total={total_time:.2f}s, "
            f"docs={len(docs)}"
        )
        
        return answer
    except Exception as e:
```

### 7.2 常见陷阱

**陷阱1：同步阻塞**
- 问题：LLM调用阻塞请求线程
- 解决：异步处理+流式输出

**陷阱2：状态泄漏**
- 问题：不同用户看到彼此的对话
- 解决：严格的会话隔离

**陷阱3：成本失控**
- 问题：无限调用LLM导致费用暴涨
- 解决：限流+缓存+降级
